1//! A local stand-in for an index service.
2//!
3//! Publishing to a real index means holding a credential for a git repository,
4//! which is a deployment's business and not a laptop's. This serves the same
5//! two routes over a plain directory so `berm publish` and `berm search` can be
6//! exercised end to end without one.
7//!
8//! It is deliberately not the real thing: no git, no credential, no identity,
9//! and it re-reads the directory on every request rather than holding it.
10
11use anyhow::{Context, Result};
12use axum::{
13    Json, Router,
14    extract::{Path as Route, Query, State},
15    http::StatusCode,
16    response::{IntoResponse, Response},
17    routing::get,
18};
19use berm_api::Failed;
20use berm_index::{Entry, Index};
21use berm_oci::{Access, Reference, Registry};
22use serde::Deserialize;
23use std::{
24    net::SocketAddr,
25    path::{Path, PathBuf},
26    str::FromStr,
27    sync::Arc,
28};

30pub async fn serve(root: PathBuf, addr: SocketAddr) -> Result<()> {
31    tokio::fs::create_dir_all(&root)
32        .await
33        .with_context(|| format!("cannot open {}", root.display()))?;
34
35    let app = Router::new()
36        .route("/berm", get(search).post(publish))
37        .route("/berm/{*key}", get(versions))
38        .with_state(Arc::new(root));
39
40    let listener = tokio::net::TcpListener::bind(addr)
41        .await
42        .with_context(|| format!("failed to bind {addr}"))?;
43    tracing::info!("listening on http://{addr}");
44    axum::serve(listener, app)
45        .with_graceful_shutdown(async {
46            let _ = tokio::signal::ctrl_c().await;
47        })
48        .await
49        .context("server failed")
50}

52#[derive(Deserialize)]
53struct Search {
54    #[serde(default)]
55    q: String,
56}
57
58async fn search(
59    State(root): State<Arc<PathBuf>>,
60    Query(search): Query<Search>,
61) -> Result<Json<Vec<Entry>>, Refused> {
62    let index = load(&root)?;
63    Ok(Json(index.search(&search.q).into_iter().cloned().collect()))
64}
65
66async fn versions(
67    State(root): State<Arc<PathBuf>>,
68    Route(key): Route<String>,
69) -> Result<Json<Vec<Entry>>, Refused> {
70    match load(&root)?.versions(&key) {
71        Some(entries) => Ok(Json(entries.to_vec())),
72        None => Err(Refused(
73            StatusCode::NOT_FOUND,
74            format!("nothing published under {key:?}"),
75        )),
76    }
77}
78
79#[derive(Deserialize)]
80struct Publish {
81    reference: String,
82}
83
84async fn publish(
85    State(root): State<Arc<PathBuf>>,
86    Json(request): Json<Publish>,
87) -> Result<Json<Entry>, Refused> {
88    let entry = record(&root, &request.reference)
89        .await
90        // The reference is the caller's: an unreadable artifact is their
91        // business, not an outage here.
92        .map_err(|error| Refused(StatusCode::BAD_REQUEST, format!("{error:#}")))?;
93    Ok(Json(entry))
94}
95
96/// Pull what the reference says it is, and append a line for it.
97async fn record(root: &Path, reference: &str) -> Result<Entry> {
98    let reference = Reference::from_str(reference)?;
99    let name = reference.to_string();
100
101    // Anonymously, as a real index would: a harness nobody can pull is not one
102    // to list.
103    let (digest, manifest) = tokio::task::spawn_blocking(move || {
104        Registry::open(&reference, Access::Read)?.describe(&reference.reference)
105    })
106    .await
107    .context("the registry lookup panicked")??;
108
109    let entry = Entry::new(name, digest, None, manifest);
110    let path = root.join(format!("{}.json", entry.key()));
111    let parent = path.parent().context("an index path has no parent")?;
112    tokio::fs::create_dir_all(parent).await?;
113
114    let mut body = tokio::fs::read_to_string(&path).await.unwrap_or_default();
115    body.push_str(&serde_json::to_string(&entry)?);
116    body.push('\n');
117    tokio::fs::write(&path, body).await?;
118    Ok(entry)
119}

121/// Read the directory per request. A stand-in has no reason to cache, and a
122/// stale answer while testing is worse than a slow one.
123fn load(root: &Path) -> Result<Index, Refused> {
124    Index::load(root).map_err(|error| {
125        Refused(
126            StatusCode::INTERNAL_SERVER_ERROR,
127            format!("cannot read the index: {error:#}"),
128        )
129    })
130}
131
132struct Refused(StatusCode, String);
133
134impl IntoResponse for Refused {
135    fn into_response(self) -> Response {
136        (self.0, Json(Failed { error: self.1 })).into_response()
137    }
