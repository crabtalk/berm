1//! Rust types to JSON Schema, at compile time.
2//!
3//! A tool's arguments are a struct the author already wrote; the schema the
4//! model reads is derived from it rather than repeated by hand in a string
5//! literal that nothing checks. Field doc comments become descriptions, and
6//! `Option<T>` is what makes a field optional.
7
8use crate::docs;
9use syn::{Fields, GenericArgument, ItemStruct, PathArguments, Type};
10
11/// Build the `parameters` object for a tool from its argument struct.
12pub fn object(item: &ItemStruct) -> syn::Result<String> {
13    let Fields::Named(fields) = &item.fields else {
14        return Err(syn::Error::new_spanned(
15            &item.ident,
16            "a tool's arguments must be a struct with named fields — the names are what the model fills in",
17        ));
18    };
19
20    let mut properties = Vec::new();
21    let mut required = Vec::new();
22
23    for field in &fields.named {
24        let name = field
25            .ident
26            .as_ref()
27            .expect("named fields have idents")
28            .to_string();
29        let (ty, optional) = unwrap_option(&field.ty);
30        let description = docs(&field.attrs);
31
32        let mut property = type_of(ty)?;
33        if !description.is_empty() {
34            // Exactly one brace: a nested type like `Vec<T>` or a map ends in
35            // two, and trimming both puts the description inside the inner
36            // object and leaves the JSON one brace short.
37            let open = property
38                .strip_suffix('}')
39                .ok_or_else(|| unsupported(&field.ty))?;
40            property = format!(
41                r#"{open},"description":"{}"}}"#,
42                crate::escape(&description)
43            );
44        }
45
46        properties.push(format!(r#""{}":{}"#, crate::escape(&name), property));
47        if !optional {
48            required.push(format!(r#""{}""#, crate::escape(&name)));
49        }
50    }
51
52    let properties = properties.join(",");
53    let required = required.join(",");
54    Ok(format!(
55        r#"{{"type":"object","properties":{{{properties}}},"required":[{required}]}}"#
56    ))
57}
58
59/// `Option<T>` means the model may omit the field; everything else is required.
60fn unwrap_option(ty: &Type) -> (&Type, bool) {
61    if let Type::Path(path) = ty
62        && let Some(segment) = path.path.segments.last()
63        && segment.ident == "Option"
64        && let PathArguments::AngleBracketed(args) = &segment.arguments
65        && let Some(GenericArgument::Type(inner)) = args.args.first()
66    {
67        return (inner, true);
68    }
69    (ty, false)
70}

72/// The JSON Schema fragment for one Rust type.
73fn type_of(ty: &Type) -> syn::Result<String> {
74    if let Type::Reference(reference) = ty {
75        return type_of(&reference.elem);
76    }
77
78    let Type::Path(path) = ty else {
79        return Err(unsupported(ty));
80    };
81    let Some(segment) = path.path.segments.last() else {
82        return Err(unsupported(ty));
83    };
84
85    let name = segment.ident.to_string();
86    if name == "Vec" {
87        let PathArguments::AngleBracketed(args) = &segment.arguments else {
88            return Err(unsupported(ty));
89        };
90        let Some(GenericArgument::Type(inner)) = args.args.first() else {
91            return Err(unsupported(ty));
92        };
93        return Ok(format!(r#"{{"type":"array","items":{}}}"#, type_of(inner)?));
94    }
95
96    // A map is an open object: the keys are the caller's, so only the values
97    // have a schema. Keys are strings because that is what JSON objects have.
98    if name == "BTreeMap" || name == "HashMap" {
99        let PathArguments::AngleBracketed(args) = &segment.arguments else {
100            return Err(unsupported(ty));
101        };
102        let mut types = args.args.iter().filter_map(|arg| match arg {
103            GenericArgument::Type(ty) => Some(ty),
104            _ => None,
105        });
106        let (Some(key), Some(value)) = (types.next(), types.next()) else {
107            return Err(unsupported(ty));
108        };
109        if type_of(key)? != r#"{"type":"string"}"# {
110            return Err(syn::Error::new_spanned(
111                key,
112                "a map's keys become JSON object keys, so they have to be strings",
113            ));
114        }
115        return Ok(format!(
116            r#"{{"type":"object","additionalProperties":{}}}"#,
117            type_of(value)?
118        ));
119    }
120
121    let primitive = match name.as_str() {
122        "String" | "str" => "string",
123        "bool" => "boolean",
124        "f32" | "f64" => "number",
125        "u8" | "u16" | "u32" | "u64" | "usize" | "i8" | "i16" | "i32" | "i64" | "isize" => {
126            "integer"
127        }
128        _ => return Err(unsupported(ty)),
129    };
130    Ok(format!(r#"{{"type":"{primitive}"}}"#))
131}
132
133fn unsupported(ty: &Type) -> syn::Error {
134    syn::Error::new_spanned(
135        ty,
136        "a tool argument must be a string, number, boolean, Vec, or Option of those — \
137         the model fills these in from a JSON Schema, and there is no schema for this type",
138    )
139}
