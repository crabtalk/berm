1//! berm.app — a workbench for harnesses.
2//!
3//! The app *is* bermd. It owns the [`Service`], binds the endpoint, and runs
4//! both on a tokio runtime beside gpui's, so launching this window starts the
5//! daemon: a `berm` CLI in a terminal and an agent's MCP client reach the same
6//! harnesses the window paints.

48/// What the app is, once it is anything.
49enum Engine {
50    Starting,
51    Serving {
52        service: Arc<Service>,
53        addr: SocketAddr,
54    },
55    /// A taken port means a second berm already holds this root — not something
56    /// to half-work through.
57    Failed(String),
58}
59
60/// Which half of a harness the pane is showing: what it claims to be, or what
61/// it does.
62#[derive(Clone, Copy, PartialEq, Eq)]
63enum Tab {
64    Tools,
65    Run,
66}
67
68/// What the pane is on: a harness this machine holds, or one the index lists.
69///
70/// The entry is held whole rather than looked up, so editing the search does
71/// not empty the pane under the person reading it.
72enum Showing {
73    Deployed(String),
74    Published(Entry),
75}
76
77/// What the index last said about the term in the field.
78enum Found {
79    /// Nothing typed, or nothing asked yet.
80    Idle,
81    Listed(Vec<Entry>),
82    /// The list could not be read at all — a clone that failed, a service that
83    /// is not there.
84    Unreachable(String),
85}

87pub struct Workbench {
88    /// Beside gpui's own executor, because `Service` is tokio: the
89    /// `spawn_blocking` a call enters the guest on panics outside a runtime.
90    rt: Runtime,
91    engine: Engine,
92    /// What the service was holding as of the last refresh.
93    harnesses: Vec<Arc<Deployed>>,
94    /// By name, not index: a refresh renumbers the list, and a harness removed
95    /// under the pane should empty it rather than hand the selection to a
96    /// neighbour.
97    selection: Option<Showing>,
98    /// The list of published harnesses, opened once — a `.git` default clones
99    /// the first time, and doing that under a keystroke would race a second
100    /// clone into the same directory.
101    index: Option<Arc<Source>>,
102    query: Entity<TextField>,
103    /// The term [`Found`] is an answer to, which is also what says whether the
104    /// rail is showing what is deployed or what is published.
105    term: String,
106    found: Found,
107    /// Bumped per edit, so an answer that comes back under an old number is
108    /// dropped rather than painted over a newer one.
109    asked: usize,
110    /// Which schemas are open, keyed the way MCP names a tool.
111    expanded: HashSet<String>,
112    tab: Tab,
113    /// The tool the Run pane is pointed at.
114    tool: Option<String>,
115    arguments: Entity<TextField>,
116    /// Every invocation this session, oldest first.
117    transcript: Vec<Invocation>,
118    /// The deploy sheet, while it is open.
119    sheet: Option<Sheet>,
120    /// The image being deployed, while it is being deployed — a path from the
121    /// sheet or a reference from the index, whichever asked.
122    deploying: Option<String>,
123    /// Whether the header is asking about removing the selected harness.
124    confirming: bool,
125    /// Why the last change to the deployed set was refused. One field for both
126    /// deploy and removal: what a person needs is the reason, and which of the
127    /// two it was is already on screen.
128    refused: Option<String>,
129    /// The window's resting focus: a press moves focus to the innermost handle
130    /// it lands in, so without one here a click off a field never takes it.
131    focus_handle: FocusHandle,
132}

134impl Workbench {
135    pub fn new(cx: &mut Context<Self>) -> Self {
136        let rt = Runtime::new().expect("failed to start a tokio runtime");
137        let starting = rt.spawn(start());
138        cx.spawn(async move |this, cx| {
139            let engine = match starting.await {
140                Ok(Ok((service, addr))) => Engine::Serving { service, addr },
141                Ok(Err(error)) => Engine::Failed(format!("{error:#}")),
142                Err(error) => Engine::Failed(format!("startup panicked: {error}")),
143            };
144            let _ = this.update(cx, |this, cx| {
145                this.engine = engine;
146                this.refresh(cx);
147                this.watch(cx);
148                cx.notify();
149            });
150        })
151        .detach();
152
153        // Beside the engine rather than after it: reading the list needs
154        // nothing from the service, and the clone it may do is slow.
155        let opening = rt.spawn_blocking(|| Source::new(None));
156        cx.spawn(async move |this, cx| {
157            let opened = match opening.await {
158                Ok(Ok(source)) => Ok(Arc::new(source)),
159                Ok(Err(error)) => Err(format!("{error:#}")),
160                Err(error) => Err(format!("opening the index panicked: {error}")),
161            };
162            let _ = this.update(cx, |this, cx| {
163                match opened {
164                    Ok(source) => {
165                        this.index = Some(source);
166                        // A term typed while this was opening was never asked.
167                        if !this.term.is_empty() {
168                            this.look(cx);
169                        }
170                    }
171                    Err(error) => this.found = Found::Unreachable(error),
172                }
173                cx.notify();
174            });
175        })
176        .detach();
177
178        let query = cx.new(|cx| TextField::new(cx).with_placeholder("Search the index…"));
179        cx.observe(&query, |this, _, cx| this.search(cx)).detach();
180
181        Self {
182            rt,
183            engine: Engine::Starting,
184            harnesses: Vec::new(),
185            selection: None,
186            index: None,
187            query,
188            term: String::new(),
189            found: Found::Idle,
190            asked: 0,
191            expanded: HashSet::new(),
192            tab: Tab::Tools,
193            tool: None,
194            arguments: cx.new(|cx| {
195                TextField::new(cx)
196                    .with_shape(Shape::Rows(8))
197                    .with_placeholder("{}")
198            }),
199            transcript: Vec::new(),
200            sheet: None,
201            deploying: None,
202            confirming: false,
203            refused: None,
204            focus_handle: cx.focus_handle(),
205        }
206    }
207
208    /// The deployed harness the pane is on, if it is on one.
209    fn selected(&self) -> Option<&Arc<Deployed>> {
210        let Some(Showing::Deployed(name)) = &self.selection else {
211            return None;
212        };
213        self.harnesses
214            .iter()
215            .find(|deployed| deployed.name == *name)
216    }
217
218    /// Show a harness, with the Run pane pointed at its first tool.
219    fn select(&mut self, name: &str, cx: &mut Context<Self>) {
220        self.selection = Some(Showing::Deployed(name.to_owned()));
221        self.tool = None;
222        self.confirming = false;
223        let first = self
224            .harnesses
225            .iter()
226            .find(|deployed| deployed.name == name)
227            .and_then(|deployed| deployed.manifest().tools.first().cloned());
228        if let Some(tool) = first {
229            self.choose(&tool, cx);
230        }
231    }
232
233    /// Re-read the deployed set.
234    fn refresh(&mut self, cx: &mut Context<Self>) {
235        let Engine::Serving { service, .. } = &self.engine else {
236            return;
237        };
238        let service = service.clone();
239        let listing = self.rt.spawn(async move { service.list().await });
240        cx.spawn(async move |this, cx| {
241            let Ok(harnesses) = listing.await else { return };
242            let _ = this.update(cx, |this, cx| {
243                this.harnesses = harnesses;
244                cx.notify();
245            });
246        })
247        .detach();
248    }
249
250    /// Follow the deployed set, which a `berm deploy` from a terminal moves
251    /// under this window. The same broadcast MCP sessions turn into
252    /// `tools/list_changed`; lagging past its backlog costs a notification, not
253    /// the watch.
254    fn watch(&self, cx: &mut Context<Self>) {
255        let Engine::Serving { service, .. } = &self.engine else {
256            return;
257        };
258        let mut changed = service.subscribe();
259        cx.spawn(async move |this, cx| {
260            while !matches!(
261                changed.recv().await,
262                Err(broadcast::error::RecvError::Closed)
263            ) {
264                if this.update(cx, |this, cx| this.refresh(cx)).is_err() {
265                    break;
266                }
267            }
268        })
269        .detach();
270    }
271
272    /// The line that is the app's reason to stay running: what to point an
273    /// agent at.
274    fn status(&self, theme: &Theme, cx: &mut Context<Self>) -> AnyElement {
275        let bar = div()
276            .flex_none()
277            .h(px(30.0))
278            .px(px(14.0))
279            .flex()
280            .flex_row()
281            .items_center()
282            .gap(px(8.0))
283            .bg(theme.surface)
284            .border_t_1()
285            .border_color(theme.border)
286            .text_size(px(11.5))
287            .text_color(theme.text_muted);
288
289        match &self.engine {
290            Engine::Starting => bar
291                .child(widgets::status_dot(theme.warning))
292                .child("starting…")
293                .into_any_element(),
294            Engine::Failed(error) => bar
295                .child(widgets::status_dot(theme.danger))
296                .child(div().text_color(theme.danger_muted).child(error.clone()))
297                .into_any_element(),
298            Engine::Serving { addr, .. } => {
299                let mcp = format!("http://{addr}/mcp");
300                bar.id("status")
301                    .cursor_pointer()
302                    .child(widgets::status_dot(theme.success))
303                    .child(format!("serving {addr} · mcp at {mcp}"))
304                    .child(
305                        div()
306                            .text_color(theme.text_faint)
307                            .child("— click to copy the endpoint"),
308                    )
309                    .on_click(cx.listener(move |_, _, _, cx| {
310                        cx.write_to_clipboard(ClipboardItem::new_string(mcp.clone()));
311                    }))
312                    .into_any_element()
313            }
314        }
315    }
316}

318impl Render for Workbench {
319    fn render(&mut self, window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
320        let theme = Theme::of(cx).clone();
321        // Hover fades paint once and stick unless a frame is asked for.
322        if motion::hover_fades_active() {
323            window.request_animation_frame();
324        }
325        // Traversal on the root, so `tab` works wherever focus happens to be.
326        focus::traversal(div())
327            .track_focus(&self.focus_handle)
328            .relative()
329            .size_full()
330            .flex()
331            .flex_col()
332            .bg(theme.bg)
333            .text_color(theme.text)
334            .font_family(theme.font_sans.clone())
335            .child(
336                div()
337                    .flex_1()
338                    .min_h_0()
339                    .flex()
340                    .flex_row()
341                    .child(self.sidebar(&theme, cx))
342                    .child(self.detail(&theme, cx)),
343            )
344            // A refusal the sheet is not already showing — a removal's, or a
345            // deploy's after the sheet closed.
346            .when(self.sheet.is_none(), |root| {
347                root.when_some(self.refused.clone(), |root, error| {
348                    root.child(
349                        div()
350                            .px(px(28.0))
351                            .pb(px(12.0))
352                            .child(theme.error_strip(error)),
353                    )
354                })
355            })
356            .child(self.status(&theme, cx))
357            .child(self.sheet(&theme, cx))
358    }
359}

361/// Open the engine and put it on the wire.
362///
363/// The listener is bound here rather than inside [`Service::serve`] so a taken
364/// port is known before anything claims to be serving.
365async fn start() -> Result<(Arc<Service>, SocketAddr)> {
366    let home = std::env::var("HOME").context("HOME is not set")?;
367    let service = Service::new(PathBuf::from(home).join(".berm")).await?;
368    let listener = TcpListener::bind(ADDR)
369        .await
370        .with_context(|| format!("failed to bind {ADDR}"))?;
371    let addr = listener
372        .local_addr()
373        .context("the listener has no address")?;
374
375    tokio::spawn(service.clone().serve(listener));
376    Ok((service, addr))
