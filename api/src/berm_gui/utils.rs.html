9/// How much of a digest reads as an identity without taking the row.
10const DIGEST_SHORT: usize = 12;
11
12/// The `sha256:` an index carries is how a registry addresses the bytes, not
13/// part of what tells two of them apart.
14pub fn short(digest: &str) -> &str {
15    let hex = digest.strip_prefix("sha256:").unwrap_or(digest);
16    &hex[..DIGEST_SHORT.min(hex.len())]
17}

19/// A published harness as a person reads it: the name its author gave the
20/// image, and the version it was pushed under.
21pub fn split(entry: &Entry) -> (&str, &str) {
22    let key = entry.key();
23    let name = key.rsplit('/').next().unwrap_or(key);
24    let version = entry.reference[key.len()..].trim_start_matches([':', '@']);
25    (name, version)
26}

28/// Read an image: a file if one is there, a registry reference otherwise. The
29/// file wins, so an image sitting in the working directory is never mistaken
30/// for something to go and fetch.
31///
32/// Blocking — the registry client is.
33pub fn image(spec: &str) -> Result<Vec<u8>> {
34    let path = Path::new(spec);
35    if path.exists() {
36        return fs::read(path).with_context(|| format!("cannot read {spec}"));
37    }
38
39    let reference = Reference::from_str(spec)
40        .with_context(|| format!("{spec:?} is neither a file nor a registry reference"))?;
41    Registry::open(&reference, Access::Read)?.pull(&reference.reference)
42}

44/// An argument object shaped like the schema advertises, with every value
45/// empty — what to fill in rather than what to look up.
46pub fn skeleton(schema: &Value) -> String {
47    let Some(properties) = schema.get("properties").and_then(Value::as_object) else {
48        return "{}".to_owned();
49    };
50    let arguments = properties
51        .iter()
52        .map(|(name, spec)| (name.clone(), empty(spec)))
53        .collect::<Map<_, _>>();
54    serde_json::to_string_pretty(&Value::Object(arguments)).unwrap_or_else(|_| "{}".to_owned())
55}
56
57fn empty(spec: &Value) -> Value {
58    match spec.get("type").and_then(Value::as_str) {
59        Some("string") => Value::String(String::new()),
60        Some("integer" | "number") => Value::Number(0.into()),
61        Some("boolean") => Value::Bool(false),
62        Some("array") => Value::Array(Vec::new()),
63        Some("object") => Value::Object(Map::new()),
64        _ => Value::Null,
65    }
66}
