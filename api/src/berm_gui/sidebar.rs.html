1//! The rail of harnesses — what is deployed, or what the index lists.
2
3use crate::{Found, Sheet, Showing, TITLEBAR, Workbench, utils};
4use bermd::Deployed;
5use bezel::{
6    gpui::{AnyElement, Context, SharedString, div, prelude::*, px},
7    theme::Theme,
8    ui::widgets::{self, ButtonStyle, Buttons, Scaffolding},
9};
10use std::sync::Arc;
11
12const WIDTH: f32 = 260.0;
13
14impl Workbench {
15    pub(crate) fn sidebar(&self, theme: &Theme, cx: &mut Context<Self>) -> AnyElement {
16        // One rail, two lists: an empty field is what this machine holds, and a
17        // term is what has been published.
18        let (heading, count) = match self.term.is_empty() {
19            true => ("Harnesses", self.harnesses.len()),
20            false => match &self.found {
21                Found::Listed(entries) => ("Index", entries.len()),
22                _ => ("Index", 0),
23            },
24        };
25
26        div()
27            .w(px(WIDTH))
28            .flex_none()
29            .h_full()
30            .flex()
31            .flex_col()
32            .bg(theme.surface)
33            .border_r_1()
34            .border_color(theme.border)
35            .child(
36                div()
37                    .px(px(20.0))
38                    .pt(px(TITLEBAR))
39                    .pb(px(10.0))
40                    .child(theme.page_header(heading, Some(count))),
41            )
42            .child(div().px(px(16.0)).pb(px(10.0)).child(self.query.clone()))
43            .child(self.list(theme, cx))
44            .child(
45                div()
46                    .flex_none()
47                    .p(px(16.0))
48                    .border_t_1()
49                    .border_color(theme.border)
50                    .child(
51                        theme
52                            .button("Deploy…", ButtonStyle::Prominent, None)
53                            .id("deploy")
54                            .w_full()
55                            .flex()
56                            .items_center()
57                            .justify_center()
58                            .on_click(cx.listener(|this, _, _, cx| {
59                                this.refused = None;
60                                this.sheet = Some(Sheet::new(cx));
61                                cx.notify();
62                            })),
63                    ),
64            )
65            .into_any_element()
66    }
67
68    fn list(&self, theme: &Theme, cx: &mut Context<Self>) -> AnyElement {
69        let rail = div().id("harnesses").flex_1().min_h_0().overflow_y_scroll();
70        if self.term.is_empty() {
71            return rail
72                .children(
73                    self.harnesses
74                        .iter()
75                        .enumerate()
76                        .map(|(index, deployed)| self.row(deployed, index == 0, theme, cx))
77                        .collect::<Vec<_>>(),
78                )
79                .into_any_element();
80        }
81
82        let note = |copy: String| {
83            div()
84                .px(px(20.0))
85                .py(px(10.0))
86                .text_size(px(12.5))
87                .text_color(theme.text_muted)
88                .child(copy)
89        };
90        match &self.found {
91            Found::Idle => rail.child(note("searching…".to_owned())),
92            Found::Unreachable(error) => {
93                rail.child(note(error.clone()).text_color(theme.danger_muted))
94            }
95            Found::Listed(entries) if entries.is_empty() => {
96                rail.child(note(format!("nothing published matches {:?}", self.term)))
97            }
98            Found::Listed(entries) => rail.children(
99                entries
100                    .iter()
101                    .enumerate()
102                    .map(|(index, entry)| self.result(entry, index == 0, theme, cx))
103                    .collect::<Vec<_>>(),
104            ),
105        }
106        .into_any_element()
107    }
108
109    fn row(
110        &self,
111        deployed: &Arc<Deployed>,
112        first: bool,
113        theme: &Theme,
114        cx: &mut Context<Self>,
115    ) -> AnyElement {
116        let name = deployed.name.clone();
117        let selected = matches!(&self.selection, Some(Showing::Deployed(at)) if *at == name);
118        let tools = deployed.manifest().tools.len();
119
120        theme
121            .card_row(first)
122            .id(SharedString::from(name.clone()))
123            .py(px(10.0))
124            .cursor_pointer()
125            .when(selected, |row| row.bg(theme.element_active))
126            .hover(widgets::card_row_hover)
127            .child(widgets::status_dot(theme.success))
128            .child(
129                div()
130                    .min_w_0()
131                    .flex()
132                    .flex_col()
133                    .child(theme.row_title(name.clone()))
134                    .child(theme.meta_line(vec![
135                        div()
136                            .child(format!("{tools} tool{}", if tools == 1 { "" } else { "s" }))
137                            .into_any_element(),
138                        div()
139                            .font_family(theme.font_mono.clone())
140                            .child(utils::short(&deployed.digest).to_owned())
141                            .into_any_element(),
142                    ])),
143            )
144            .on_click(cx.listener(move |this, _, _, cx| {
145                this.select(&name, cx);
146                cx.notify();
147            }))
148            .into_any_element()
149    }
150}
