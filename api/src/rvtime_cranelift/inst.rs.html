6/// The CLIF comparison for a RISC-V branch condition.
7pub fn condition(op: Cond) -> IntCC {
8    match op {
9        Cond::Eq => IntCC::Equal,
10        Cond::Ne => IntCC::NotEqual,
11        Cond::Lt => IntCC::SignedLessThan,
12        Cond::Ge => IntCC::SignedGreaterThanOrEqual,
13        Cond::LtU => IntCC::UnsignedLessThan,
14        Cond::GeU => IntCC::UnsignedGreaterThanOrEqual,
15    }
16}

18/// Translate a guest address into a host one.
19///
20/// The mask is the sandbox. Guest addresses are 64-bit values a program is free
21/// to compute arbitrarily; `and`-ing with `size - 1` keeps every access inside
22/// the reserved window, where anything not committed is a guard page. Without
23/// it a guest could compute an address past the reservation and read host
24/// memory.
25///
26/// This confines rather than bounds-checks: an out-of-range address wraps and
27/// may land on a committed page instead of faulting. That is enough for a
28/// sandbox -- the guest can only reach its own memory -- but it is not a
29/// precise bounds check, and it is why the size must be a power of two.
30pub fn address(b: &mut FunctionBuilder, memory: Value, base: Value, imm: i64, mask: i64) -> Value {
31    let addr = b.ins().iadd_imm_s(base, imm);
32    let masked = b.ins().band_imm_u(addr, mask);
33    b.ins().iadd(memory, masked)
34}

36/// Flags for guest accesses.
37///
38/// Deliberately not `MemFlagsData::trusted()`: these accesses *can* trap, which is
39/// how bounds checking works, and RISC-V permits unaligned access.
40fn guest_access() -> MemFlagsData {
41    MemFlagsData::new()
42}

44/// Emit a load, extending to 64 bits as the opcode requires.
45pub fn load(
46    b: &mut FunctionBuilder,
47    memory: Value,
48    op: LoadOp,
49    base: Value,
50    imm: i64,
51    mask: i64,
52) -> Value {
53    let addr = address(b, memory, base, imm, mask);
54    let flags = guest_access();
55
56    match op {
57        LoadOp::B => {
58            let v = b.ins().load(types::I8, flags, addr, 0);
59            b.ins().sextend(types::I64, v)
60        }
61        LoadOp::H => {
62            let v = b.ins().load(types::I16, flags, addr, 0);
63            b.ins().sextend(types::I64, v)
64        }
65        LoadOp::W => {
66            let v = b.ins().load(types::I32, flags, addr, 0);
67            b.ins().sextend(types::I64, v)
68        }
69        LoadOp::D => b.ins().load(types::I64, flags, addr, 0),
70        LoadOp::Bu => {
71            let v = b.ins().load(types::I8, flags, addr, 0);
72            b.ins().uextend(types::I64, v)
73        }
74        LoadOp::Hu => {
75            let v = b.ins().load(types::I16, flags, addr, 0);
76            b.ins().uextend(types::I64, v)
77        }
78        LoadOp::Wu => {
79            let v = b.ins().load(types::I32, flags, addr, 0);
80            b.ins().uextend(types::I64, v)
81        }
82    }
83}

85/// Emit a store, truncating to the opcode's width.
86pub fn store(
87    b: &mut FunctionBuilder,
88    memory: Value,
89    op: StoreOp,
90    base: Value,
91    imm: i64,
92    value: Value,
93    mask: i64,
94) {
95    let addr = address(b, memory, base, imm, mask);
96    let flags = guest_access();
97
98    let narrowed = match op {
99        StoreOp::B => b.ins().ireduce(types::I8, value),
100        StoreOp::H => b.ins().ireduce(types::I16, value),
101        StoreOp::W => b.ins().ireduce(types::I32, value),
102        StoreOp::D => value,
103    };
104    b.ins().store(flags, narrowed, addr, 0);
105}

107/// Emit a base integer ALU operation.
108///
109/// `word` selects the `*w` forms, which compute on the low 32 bits and
110/// sign-extend the result back to 64.
111pub fn alu(b: &mut FunctionBuilder, op: AluOp, lhs: Value, rhs: Value, word: bool) -> Value {
112    if word {
113        let lhs = b.ins().ireduce(types::I32, lhs);
114        let rhs = b.ins().ireduce(types::I32, rhs);
115        let result = alu_typed(b, op, lhs, rhs, types::I32);
116        return b.ins().sextend(types::I64, result);
117    }
118    alu_typed(b, op, lhs, rhs, types::I64)
119}
120
121fn alu_typed(b: &mut FunctionBuilder, op: AluOp, lhs: Value, rhs: Value, ty: types::Type) -> Value {
122    match op {
123        AluOp::Add => b.ins().iadd(lhs, rhs),
124        AluOp::Sub => b.ins().isub(lhs, rhs),
125        // RISC-V masks the shift amount to the operand width, which is exactly
126        // what CLIF's shifts already do.
127        AluOp::Sll => b.ins().ishl(lhs, rhs),
128        AluOp::Srl => b.ins().ushr(lhs, rhs),
129        AluOp::Sra => b.ins().sshr(lhs, rhs),
130        AluOp::Xor => b.ins().bxor(lhs, rhs),
131        AluOp::Or => b.ins().bor(lhs, rhs),
132        AluOp::And => b.ins().band(lhs, rhs),
133        AluOp::Slt => {
134            let flag = b.ins().icmp(IntCC::SignedLessThan, lhs, rhs);
135            b.ins().uextend(ty, flag)
136        }
137        AluOp::SltU => {
138            let flag = b.ins().icmp(IntCC::UnsignedLessThan, lhs, rhs);
139            b.ins().uextend(ty, flag)
140        }
141    }
142}

144/// Emit an M-extension operation.
145///
146/// RISC-V defines division by zero and signed overflow to produce specific
147/// values rather than trapping, so the divisor is sanitised first and the
148/// special results are selected afterwards. CLIF's `sdiv`/`udiv` trap on those
149/// inputs, so this cannot be left to the hardware.
150pub fn muldiv(b: &mut FunctionBuilder, op: MulOp, lhs: Value, rhs: Value, word: bool) -> Value {
151    if word {
152        let lhs = b.ins().ireduce(types::I32, lhs);
153        let rhs = b.ins().ireduce(types::I32, rhs);
154        let result = muldiv_typed(b, op, lhs, rhs, types::I32);
155        return b.ins().sextend(types::I64, result);
156    }
157    muldiv_typed(b, op, lhs, rhs, types::I64)
158}
159
160fn muldiv_typed(
161    b: &mut FunctionBuilder,
162    op: MulOp,
163    lhs: Value,
164    rhs: Value,
165    ty: types::Type,
166) -> Value {
167    let min = if ty == types::I32 {
168        i32::MIN as i64
169    } else {
170        i64::MIN
171    };
172
173    match op {
174        MulOp::Mul => b.ins().imul(lhs, rhs),
175        MulOp::MulH => b.ins().smulhi(lhs, rhs),
176        MulOp::MulHU => b.ins().umulhi(lhs, rhs),
177        // mulhsu has no CLIF equivalent. Treating the signed operand as
178        // unsigned differs from the true product by 2^N * rhs exactly when the
179        // operand is negative, so correcting the unsigned high half by `rhs`
180        // in that case gives the signed-by-unsigned result.
181        MulOp::MulHSU => {
182            let high = b.ins().umulhi(lhs, rhs);
183            let negative = b.ins().icmp_imm_s(IntCC::SignedLessThan, lhs, 0);
184            let zero = b.ins().iconst(ty, 0);
185            let correction = b.ins().select(negative, rhs, zero);
186            b.ins().isub(high, correction)
187        }
188
189        MulOp::Div => {
190            let (safe, by_zero, overflow) = sanitise(b, lhs, rhs, ty, min);
191            let quotient = b.ins().sdiv(lhs, safe);
192            let min_value = b.ins().iconst(ty, min);
193            let quotient = b.ins().select(overflow, min_value, quotient);
194            let all_ones = b.ins().iconst(ty, -1);
195            b.ins().select(by_zero, all_ones, quotient)
196        }
197        MulOp::Rem => {
198            let (safe, by_zero, overflow) = sanitise(b, lhs, rhs, ty, min);
199            let remainder = b.ins().srem(lhs, safe);
200            let zero = b.ins().iconst(ty, 0);
201            let remainder = b.ins().select(overflow, zero, remainder);
202            b.ins().select(by_zero, lhs, remainder)
203        }
204        MulOp::DivU => {
205            let by_zero = b.ins().icmp_imm_s(IntCC::Equal, rhs, 0);
206            let one = b.ins().iconst(ty, 1);
207            let safe = b.ins().select(by_zero, one, rhs);
208            let quotient = b.ins().udiv(lhs, safe);
209            let all_ones = b.ins().iconst(ty, -1);
210            b.ins().select(by_zero, all_ones, quotient)
211        }
212        MulOp::RemU => {
213            let by_zero = b.ins().icmp_imm_s(IntCC::Equal, rhs, 0);
214            let one = b.ins().iconst(ty, 1);
215            let safe = b.ins().select(by_zero, one, rhs);
216            let remainder = b.ins().urem(lhs, safe);
217            b.ins().select(by_zero, lhs, remainder)
218        }
219    }
220}

222/// Replace a divisor that would trap with 1, reporting why.
223fn sanitise(
224    b: &mut FunctionBuilder,
225    lhs: Value,
226    rhs: Value,
227    ty: types::Type,
228    min: i64,
229) -> (Value, Value, Value) {
230    let by_zero = b.ins().icmp_imm_s(IntCC::Equal, rhs, 0);
231    let is_min = b.ins().icmp_imm_s(IntCC::Equal, lhs, min);
232    let is_neg_one = b.ins().icmp_imm_s(IntCC::Equal, rhs, -1);
233    let overflow = b.ins().band(is_min, is_neg_one);
234
235    let unsafe_divisor = b.ins().bor(by_zero, overflow);
236    let one = b.ins().iconst(ty, 1);
237    let safe = b.ins().select(unsafe_divisor, one, rhs);
238
239    (safe, by_zero, overflow)
240}

242/// The CLIF type for an atomic width.
243fn atomic_type(width: Width) -> types::Type {
244    match width {
245        Width::W => types::I32,
246        Width::D => types::I64,
247    }
248}

250/// Emit an atomic read-modify-write.
251///
252/// The guest is single-threaded, so these lower to a plain load, compute and
253/// store. Nothing can observe the sequence as non-atomic.
254pub fn amo(
255    b: &mut FunctionBuilder,
256    memory: Value,
257    op: AmoOp,
258    width: Width,
259    base: Value,
260    operand: Value,
261    mask: i64,
262) -> Value {
263    let ty = atomic_type(width);
264    let addr = address(b, memory, base, 0, mask);
265    let flags = guest_access();
266
267    let old = b.ins().load(ty, flags, addr, 0);
268    let operand = if ty == types::I32 {
269        b.ins().ireduce(types::I32, operand)
270    } else {
271        operand
272    };
273
274    let new = match op {
275        AmoOp::Swap => operand,
276        AmoOp::Add => b.ins().iadd(old, operand),
277        AmoOp::Xor => b.ins().bxor(old, operand),
278        AmoOp::And => b.ins().band(old, operand),
279        AmoOp::Or => b.ins().bor(old, operand),
280        AmoOp::Min => b.ins().smin(old, operand),
281        AmoOp::Max => b.ins().smax(old, operand),
282        AmoOp::MinU => b.ins().umin(old, operand),
283        AmoOp::MaxU => b.ins().umax(old, operand),
284    };
285    b.ins().store(flags, new, addr, 0);
286
287    // The architecture sign-extends the 32-bit forms' result.
288    if ty == types::I32 {
289        b.ins().sextend(types::I64, old)
290    } else {
291        old
292    }
293}

295/// Emit `lr.{w,d}`.
296pub fn atomic_load(
297    b: &mut FunctionBuilder,
298    memory: Value,
299    width: Width,
300    base: Value,
301    mask: i64,
302) -> Value {
303    let ty = atomic_type(width);
304    let addr = address(b, memory, base, 0, mask);
305    let value = b.ins().load(ty, guest_access(), addr, 0);
306    if ty == types::I32 {
307        b.ins().sextend(types::I64, value)
308    } else {
309        value
310    }
311}

313/// Emit the store half of `sc.{w,d}`.
314pub fn atomic_store(
315    b: &mut FunctionBuilder,
316    memory: Value,
317    width: Width,
318    base: Value,
319    value: Value,
320    mask: i64,
321) {
322    let ty = atomic_type(width);
323    let addr = address(b, memory, base, 0, mask);
324    let narrowed = if ty == types::I32 {
325        b.ins().ireduce(types::I32, value)
326    } else {
327        value
328    };
329    b.ins().store(guest_access(), narrowed, addr, 0);
