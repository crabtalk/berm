1//! The wire between a harness and its host.
2//!
3//! A system harness is identified by its *name*; the number `ecall` carries in
4//! `a7` is derived from it (RFC 0205). Adding a system harness therefore cannot
5//! collide with an allocation someone else made, and there is no registry of
6//! integers to keep.
7//!
8//! The same hash is computed guest-side in `crates/lang/src/abi.rs`. The two
9//! must agree — and cannot drift quietly if they don't: a mismatched name
10//! hashes to a number no closure is registered for, so the first call traps
11//! as an unknown host call rather than reaching the wrong system harness.
12
13/// Write a UTF-8 message to the host log. `(ptr, len) -> 0`
14pub const HOST_LOG: u64 = hash("berm.log");
15/// Byte length of this invocation's argument blob. `() -> len`
16pub const HOST_ARG_LEN: u64 = hash("berm.args.len");
17/// Copy the argument blob into guest memory. `(ptr, cap) -> full length`
18pub const HOST_ARG_READ: u64 = hash("berm.args.read");
19/// Fail this invocation with a message. `(ptr, len) -> 0`
20pub const HOST_FAIL: u64 = hash("berm.fail");
21/// Copy the staged system harness result into guest memory. `(ptr, cap) -> full length`
22pub const HOST_RESULT_READ: u64 = hash("berm.result.read");
23
24/// Set on a staged length when the bytes are an error message rather than a
25/// result. A length never reaches this bit on its own, so one return value
26/// carries the outcome and the size together.
27pub const ERROR: u64 = 1 << 63;
28
29/// FNV-1a over the system harness's name, evaluated at compile time.
30pub const fn hash(name: &str) -> u64 {
31    let bytes = name.as_bytes();
32    let mut result: u64 = 0xcbf2_9ce4_8422_2325;
33    let mut at = 0;
34    while at < bytes.len() {
35        result ^= bytes[at] as u64;
36        result = result.wrapping_mul(0x0000_0100_0000_01b3);
37        at += 1;
38    }
39    result
40}

42/// ELF section carrying the harness's manifest: ABI version, tools, and the
43/// system harnesses it wants. A section rather than an export, so reading what a
44/// harness claims to be never means running it.
45pub const ABI_SECTION: &str = ".berm.abi";
46/// Where this guest's heap starts. `() -> address`
47pub const HOST_HEAP_START: u64 = hash("berm.heap.start");
48/// How many bytes of it there are. `() -> length`
49pub const HOST_HEAP_SIZE: u64 = hash("berm.heap.size");
50/// Prefix on every tool's exported symbol. A tool is resolved by name like any
51/// other symbol; the prefix keeps one called `init` from colliding with the
52/// exports the ABI reserves.
53pub const TOOL_PREFIX: &str = "berm_tool_";
