1//! A bound on how long a guest may run.
2//!
3//! rvtime's interrupt check covers every way guest code can fail to terminate:
4//! a guest can only run forever by looping, and unbounded recursion exhausts
5//! the native stack and traps. What it needs is someone to ask, from a thread
6//! that is not the one blocked inside the guest.
7//!
8//! One thread serves every invocation. A thread per call would cost more to
9//! spawn than a whole invocation costs to run — the boundary is ~17µs — so
10//! deadlines are registered in a shared list and the watchdog sleeps until the
11//! earliest one, or indefinitely when there are none.
12
13use rvtime::Interrupt;
14use std::{
15    sync::{
16        Condvar, LazyLock, Mutex,
17        atomic::{AtomicU64, Ordering},
18    },
19    thread,
20    time::{Duration, Instant},
21};
22
23/// How long a guest may run before it is asked to stop.
24///
25/// A guest blocked in a host call cannot notice an interrupt until that call
26/// returns, so an embedder's system harness has to time out well inside this.
27/// The bound exists to stop non-termination, not to enforce latency, and a
28/// harness doing slow but finite work should finish rather than be killed.
29const TIMEOUT: Duration = Duration::from_secs(60);
30
31/// One guest's deadline: the ticket that withdraws it, when it expires, and
32/// the handle that stops it.
33struct Entry {
34    ticket: u64,
35    at: Instant,
36    interrupt: Interrupt,
37}
38
39/// Deadlines the watchdog is waiting on, keyed by a monotonic ticket so a
40/// finished invocation removes its own rather than someone else's.
41static PENDING: LazyLock<(Mutex<Vec<Entry>>, Condvar)> =
42    LazyLock::new(|| (Mutex::new(Vec::new()), Condvar::new()));
43
44/// Hands out tickets. Wrapping after 2^64 invocations is not a scenario.
45static NEXT: AtomicU64 = AtomicU64::new(0);
46
47/// Registers a deadline and withdraws it when dropped.
48///
49/// Withdrawal is the point: the overwhelming majority of invocations finish
50/// long before the deadline, and a guard that forgot to deregister would leave
51/// the watchdog interrupting a store that had already been dropped.
52pub struct Deadline(u64);
53
54impl Deadline {
55    /// Ask for `interrupt` to be tripped if this guard is still alive in
56    /// [`TIMEOUT`].
57    pub fn set(interrupt: Interrupt) -> Self {
58        let ticket = NEXT.fetch_add(1, Ordering::Relaxed);
59        let (pending, wake) = &*PENDING;
60        pending.lock().expect("watchdog deadlines").push(Entry {
61            ticket,
62            at: Instant::now() + TIMEOUT,
63            interrupt,
64        });
65        // The watchdog may be parked with no deadline to wait on, or waiting
66        // on one later than this.
67        wake.notify_one();
68
69        start();
70        Self(ticket)
71    }
72}
73
74impl Drop for Deadline {
75    fn drop(&mut self) {
76        let (pending, _) = &*PENDING;
77        pending
78            .lock()
79            .expect("watchdog deadlines")
80            .retain(|entry| entry.ticket != self.0);
81    }
82}
83
84/// Start the watchdog once, on the first invocation that needs it. An embedder
85/// that never runs a guest never gets the thread.
86fn start() {
87    static STARTED: std::sync::Once = std::sync::Once::new();
88    STARTED.call_once(|| {
89        thread::Builder::new()
90            .name("berm-watchdog".to_owned())
91            .spawn(watch)
92            .expect("spawn the berm watchdog");
93    });
94}
95
96fn watch() {
97    let (pending, wake) = &*PENDING;
98    let mut deadlines = pending.lock().expect("watchdog deadlines");
99    loop {
100        let now = Instant::now();
101        // Tripping an interrupt does not remove the deadline: the guest stops
102        // at its next backward edge rather than immediately, and its own guard
103        // is what withdraws the entry.
104        let mut earliest: Option<Instant> = None;
105        for entry in deadlines.iter() {
106            if entry.at <= now {
107                entry.interrupt.interrupt();
108            } else {
109                earliest = Some(earliest.map_or(entry.at, |e: Instant| e.min(entry.at)));
110            }
111        }
112
113        deadlines = match earliest {
114            Some(at) => {
115                wake.wait_timeout(deadlines, at.saturating_duration_since(now))
116                    .expect("watchdog deadlines")
117                    .0
118            }
119            None => wake.wait(deadlines).expect("watchdog deadlines"),
120        };
121    }
122}
