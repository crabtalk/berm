1//! The framing a system harness request and its reply share.
2//!
3//! The guest builds these in `crates/lang/src/wire.rs`: a flat sequence of
4//! `u32`-prefixed fields, one rule for every system harness. This is the only
5//! decoder, so a malformed request is rejected in one place rather than in
6//! each system harness's own hand-rolled parse — which is why it is public: an
7//! embedder supplying a [`crate::Harness`] frames it the same way rather
8//! than inventing a second layout.
9
10use anyhow::{Result, bail};
11
12/// Split a request into its fields.
13pub fn fields(mut request: &[u8]) -> Result<Vec<&[u8]>> {
14    let mut fields = Vec::new();
15    while !request.is_empty() {
16        let Some((header, rest)) = request.split_at_checked(4) else {
17            bail!("truncated field header");
18        };
19        let length = u32::from_le_bytes(header.try_into()?) as usize;
20        let Some((field, rest)) = rest.split_at_checked(length) else {
21            bail!(
22                "field claims {length} bytes but the request has {}",
23                rest.len()
24            );
25        };
26        fields.push(field);
27        request = rest;
28    }
29    Ok(fields)
30}
31
32/// Lay fields out for the guest to read back. The other direction of
33/// [`fields`], for a system harness whose answer is more than one blob.
34pub fn frame(fields: &[&[u8]]) -> Vec<u8> {
35    let mut framed = Vec::new();
36    for field in fields {
37        framed.extend_from_slice(&(field.len() as u32).to_le_bytes());
38        framed.extend_from_slice(field);
39    }
40    framed
41}
42
43/// Read a field as UTF-8.
44pub fn text<'a>(fields: &[&'a [u8]], at: usize, what: &str) -> Result<&'a str> {
45    let Some(field) = fields.get(at) else {
46        bail!("request has no {what}");
47    };
48    Ok(std::str::from_utf8(field)?)
49}
