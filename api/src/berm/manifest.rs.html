1//! What a harness says it is.
2//!
3//! Carried as an ELF section rather than an export, so reading it never means
4//! running the guest (RFC 0205). It is parsed once at load: the tool schemas
5//! the model sees come from here, and so does the usage an embedder puts in
6//! front of a model before it chooses between them.
7
8use anyhow::{Context, Result, bail};
9use object::{Object, ObjectSection};
10use serde::Deserialize;
11
12use crate::abi;
13
14/// The ABI this host speaks. A harness built against a different one is
15/// refused rather than dispatched into a system harness its author did not
16/// mean.
17pub const ABI_VERSION: u32 = 0;
18
19#[derive(Debug, Clone, Deserialize)]
20pub struct Manifest {
21    pub abi_version: u32,
22    pub tools: Vec<ToolSpec>,
23    /// When to reach for these tools, and how they go together — the
24    /// question no single tool's `description` answers, because it is about
25    /// choosing between them. An embedder puts this in front of a model
26    /// before it decides, so it is paid on every turn: a few lines, not a
27    /// manual.
28    #[serde(default)]
29    pub usage: String,
30}
31
32impl Manifest {
33    /// Pull the manifest out of the ELF. This runs before anything is compiled,
34    /// let alone executed — a harness gets to describe itself without being given
35    /// a turn.
36    /// Read what an ELF claims to be, without compiling or running it.
37    ///
38    /// This is what the section is *for* (RFC 0205): learning a harness's tools,
39    /// wants, and usage must not mean instantiating it. An embedder assembling a
40    /// prompt or listing a registry needs exactly this and nothing else.
41    pub fn from_elf(elf: &[u8]) -> Result<Self> {
42        let file = object::File::parse(elf).context("harness is not a readable ELF")?;
43        let section = file
44            .section_by_name(abi::ABI_SECTION)
45            .with_context(|| format!("harness has no {} section", abi::ABI_SECTION))?;
46        let bytes = section.data().context("harness manifest is unreadable")?;
47        let json = String::from_utf8(bytes.to_vec()).context("harness manifest is not UTF-8")?;
48        Self::parse(&json)
49    }
50
51    pub fn parse(json: &str) -> Result<Self> {
52        let manifest: Manifest =
53            serde_json::from_str(json).context("harness manifest is not valid JSON")?;
54        if manifest.abi_version != ABI_VERSION {
55            bail!(
56                "harness was built against ABI version {}, this host speaks {ABI_VERSION}",
57                manifest.abi_version
58            );
59        }
60        Ok(manifest)
61    }
62}

64#[derive(Debug, Clone, Deserialize)]
65pub struct ToolSpec {
66    pub name: String,
67    pub description: String,
68    /// JSON Schema for the tool's arguments, as the model receives it.
69    pub parameters: serde_json::Value,
70}
