1//! Registering host functions
2
3use crate::{Caller, Engine, Instance, Module, Store, abi::Regs};
4use anyhow::{Context, Result};
5use std::{collections::HashMap, sync::Arc};
6
7/// A host function, erased down to the register file it operates on.
8pub type HostFn<T> = Box<dyn Fn(Caller<'_, T>) -> Result<()> + Send + Sync>;
9
10pub(crate) type HostMap<T> = HashMap<u64, HostFn<T>>;
11
12/// Defines the host functions a guest can call.
13///
14/// A guest reaches these with `ecall`, taking the call number from `a7` and
15/// arguments from `a0` onwards -- the standard RISC-V syscall convention. The
16/// key is therefore a number rather than a name: unlike WebAssembly, an ELF
17/// carries no symbolic import table for the host to resolve.
18pub struct Linker<T> {
19    hosts: Arc<HostMap<T>>,
20}
21
22impl<T> std::fmt::Debug for Linker<T> {
23    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
24        let mut numbers: Vec<_> = self.hosts.keys().copied().collect();
25        numbers.sort_unstable();
26        f.debug_struct("Linker").field("calls", &numbers).finish()
27    }
28}
29
30impl<T> Linker<T> {
31    /// A linker with no host functions.
32    pub fn new(_engine: &Engine) -> Self {
33        Linker {
34            hosts: Arc::new(HashMap::new()),
35        }
36    }
37
38    /// Register a host function that works on registers directly.
39    ///
40    /// Use this when the call does not have a fixed arity -- reading a
41    /// variable number of arguments, say. [`Linker::func_wrap`] is nicer
42    /// otherwise.
43    pub fn func(
44        &mut self,
45        number: u64,
46        func: impl Fn(Caller<'_, T>) -> Result<()> + Send + Sync + 'static,
47    ) -> Result<&mut Self> {
48        self.insert(number, Box::new(func))
49    }
50
51    /// Register a host function with a typed signature.
52    ///
53    /// ```ignore
54    /// linker.func_wrap(1, |_: Caller<'_, ()>, a: u64, b: u64| Ok(a + b))?;
55    /// ```
56    pub fn func_wrap<P, R>(
57        &mut self,
58        number: u64,
59        func: impl IntoHostFunc<T, P, R>,
60    ) -> Result<&mut Self> {
61        self.insert(number, func.into_host())
62    }
63
64    /// Instantiate `module` into `store`.
65    ///
66    /// This maps the guest's memory and wires the store to the compiled code.
67    pub fn instantiate(&self, store: &mut Store<T>, module: &Module) -> Result<Instance> {
68        store.instantiate(module.inner().clone(), self.hosts.clone())?;
69        Ok(Instance::new(module.inner().clone()))
70    }
71
72    fn insert(&mut self, number: u64, func: HostFn<T>) -> Result<&mut Self> {
73        Arc::get_mut(&mut self.hosts)
74            .context("host functions cannot be registered after instantiate")?
75            .insert(number, func);
76        Ok(self)
77    }
78}
79
80/// Converts a closure into a [`HostFn`] by reading its arguments out of the
81/// argument registers and writing its results back.
82pub trait IntoHostFunc<T, P, R> {
83    /// Erase the closure's signature.
84    fn into_host(self) -> HostFn<T>;
85}
86
87macro_rules! host_func {
88    ($($name:ident : $ty:ty = $index:expr),*) => {
89        impl<T, F, R> IntoHostFunc<T, ($($ty,)*), R> for F
90        where
91            F: Fn(Caller<'_, T>, $($ty),*) -> Result<R> + Send + Sync + 'static,
92            R: Regs,
93        {
94            fn into_host(self) -> HostFn<T> {
95                Box::new(move |mut caller: Caller<'_, T>| {
96                    $(let $name: $ty = caller.arg($index);)*
97                    let results = self(caller.reborrow(), $($name),*)?;
98                    caller.set_results(results);
99                    Ok(())
100                })
101            }
102        }
103    };
104}
