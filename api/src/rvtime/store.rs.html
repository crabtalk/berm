16/// Why a guest stopped short.
17#[derive(Debug)]
18pub enum Trap {
19    /// A load or store landed outside the committed guest memory.
20    MemoryFault {
21        /// The guest address, when the fault was inside the address space.
22        address: Option<u64>,
23    },
24
25    /// An indirect jump targeted something that is not a function entry.
26    BadIndirectTarget {
27        /// The address the guest tried to reach.
28        target: u64,
29    },
30
31    /// The guest executed `ebreak`.
32    Breakpoint,
33
34    /// The guest called a number no host function is registered for.
35    UnknownHostCall(u64),
36
37    /// A host function returned an error.
38    HostCall(anyhow::Error),
39
40    /// The guest reached an instruction compilers place where control must not
41    /// go -- typically a panic path.
42    IllegalInstruction,
43
44    /// The host asked the guest to stop, and it did.
45    Interrupted,
46}
47
48impl fmt::Display for Trap {
49    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
50        match self {
51            Trap::MemoryFault {
52                address: Some(addr),
53            } => write!(f, "guest memory fault at {addr:#x}"),
54            Trap::MemoryFault { address: None } => {
55                write!(f, "memory fault outside the guest address space")
56            }
57            Trap::BadIndirectTarget { target } => {
58                write!(f, "indirect jump to an unknown target {target:#x}")
59            }
60            Trap::Breakpoint => write!(f, "guest executed ebreak"),
61            Trap::UnknownHostCall(number) => write!(f, "no host function for call {number}"),
62            Trap::HostCall(error) => write!(f, "host call failed: {error}"),
63            Trap::IllegalInstruction => write!(f, "guest reached an illegal instruction"),
64            Trap::Interrupted => write!(f, "guest was interrupted"),
65        }
66    }
67}

71/// Everything a running guest owns.
72pub(crate) struct State<T> {
73    pub module: Arc<compiler::Module>,
74    pub memory: Memory,
75    pub ctx: VmCtx,
76    pub hosts: Arc<HostMap<T>>,
77
78    /// The flag compiled code polls on backward edges. Held in an `Arc` so an
79    /// [`Interrupt`] handle stays valid however the store moves.
80    pub interrupt: Arc<AtomicU64>,
81
82    /// Set by a failing host call and taken by the entry point, so the error
83    /// survives the return through compiled code.
84    pub failure: Option<Trap>,
85}

87/// Owns the guest's memory, registers, and the embedder's own data.
88///
89/// A store holds a single instance. That is narrower than wasmtime, where one
90/// store can back many instances, and it is what a program image actually
91/// needs: one address space, one register file.
92pub struct Store<T> {
93    data: T,
94    engine: Engine,
95    config: Config,
96    pub(crate) state: Option<State<T>>,
97}
98
99// Every pointer reachable from a store is owned by it: `memory` is its own
100// mapping, `dispatch` and `trampoline` belong to the `Arc<Module>` it holds,
101// and `host_data` is rewritten on each entry. Moving the store moves all of
102// them together, so an embedder can hand one to another thread -- which it must
103// be able to do to run guests on a worker pool.
104//
105// Deliberately *not* `Sync`. Entering a guest takes `&mut Store`, so two
106// threads must never be inside one at the same time; `Send` without `Sync` is
107// exactly that rule.
108unsafe impl<T: Send> Send for Store<T> {}
109
110impl<T> Store<T> {
111    /// Create a store holding `data`.
112    pub fn new(engine: &Engine, data: T) -> Self {
113        Store {
114            data,
115            config: engine.config().clone(),
116            engine: engine.clone(),
117            state: None,
118        }
119    }
120
121    /// The embedder's data.
122    pub fn data(&self) -> &T {
123        &self.data
124    }
125
126    /// The embedder's data, mutably.
127    pub fn data_mut(&mut self) -> &mut T {
128        &mut self.data
129    }
130
131    /// Consume the store, returning the embedder's data.
132    pub fn into_data(self) -> T {
133        self.data
134    }
135
136    /// The engine this store was created with.
137    pub fn engine(&self) -> &Engine {
138        &self.engine
139    }
140
141    /// A handle that stops this guest from another thread.
142    ///
143    /// Requires [`Config::interruptible`](crate::Config::interruptible); without
144    /// it the compiled code has no checks to notice the request, so asking for a
145    /// handle that could never work is an error rather than a silent no-op.
146    pub fn interrupt_handle(&self) -> Result<Interrupt> {
147        let state = self.instantiated()?;
148        if !state.module.interruptible() {
149            bail!("this module was not compiled with Config::interruptible");
150        }
151        Ok(Interrupt(state.interrupt.clone()))
152    }
153
154    /// The guest heap: committed, zeroed, and guarded from the stack.
155    ///
156    /// rvtime commits the region but does not carve it up. Hand these bounds to
157    /// the guest -- through a host function you register, or as arguments to an
158    /// init export -- and let its allocator manage them.
159    pub fn heap(&self) -> Result<std::ops::Range<u64>> {
160        Ok(self.instantiated()?.memory.heap())
161    }
162
163    /// Read `len` bytes of guest memory at `addr`.
164    pub fn read(&self, addr: u64, len: u64) -> Result<&[u8]> {
165        self.instantiated()?.memory.read(addr, len)
166    }
167
168    /// Write `data` into guest memory at `addr`.
169    pub fn write(&mut self, addr: u64, data: &[u8]) -> Result<()> {
170        let state = self
171            .state
172            .as_mut()
173            .ok_or_else(|| anyhow!("store has no instance"))?;
174        state.memory.write(addr, data)
175    }
176
177    fn instantiated(&self) -> Result<&State<T>> {
178        self.state
179            .as_ref()
180            .ok_or_else(|| anyhow!("store has no instance; call Linker::instantiate first"))
181    }
182
183    /// Wire up an instance. Called by [`crate::Linker::instantiate`].
184    pub(crate) fn instantiate(
185        &mut self,
186        module: Arc<compiler::Module>,
187        hosts: Arc<HostMap<T>>,
188    ) -> Result<()> {
189        // The size comes from the module, not this store's config: the address
190        // mask is baked into the compiled code, so mapping at any other size
191        // would let a guest address escape the reservation.
192        let memory = Memory::new(module.program(), module.memory_size(), self.stack_size())
193            .context("failed to map guest memory")?;
194
195        let interrupt = Arc::new(AtomicU64::new(0));
196        let ctx = VmCtx {
197            regs: [0; 32],
198            memory: memory.base(),
199            dispatch: module.dispatch().as_ptr(),
200            dispatch_len: module.dispatch().len() as u64,
201            text_base: module.program().text.start,
202            host_call: dispatch::<T> as *const u8,
203            // Refreshed on every entry, because the store may have moved.
204            host_data: std::ptr::null_mut(),
205            interrupt: Arc::as_ptr(&interrupt).cast(),
206            trap: 0,
207            detail: 0,
208        };
209
210        self.state = Some(State {
211            module,
212            memory,
213            ctx,
214            hosts,
215            interrupt,
216            failure: None,
217        });
218        Ok(())
219    }
220
221    /// The configured stack size, rounded up to a whole number of host pages.
222    fn stack_size(&self) -> u64 {
223        let page = compiler::memory::host_page();
224        self.config.stack_size.div_ceil(page).max(1) * page
225    }
226}

228/// Call a compiled guest function.
229///
230/// `params` land in the argument registers and the results are read back out
231/// of them once the guest returns.
232pub(crate) fn enter<T, P: Regs, R: Regs>(
233    store: &mut Store<T>,
234    entry: *const u8,
235    params: P,
236) -> Result<R> {
237    // Taken before the state borrow: compiled code reaches the store through
238    // this, and the store may have moved since it was instantiated.
239    let host_data: *mut Store<T> = store;
240
241    let state = store
242        .state
243        .as_mut()
244        .ok_or_else(|| anyhow!("store has no instance; call Linker::instantiate first"))?;
245
246    state.failure = None;
247    state.ctx.trap = 0;
248    state.ctx.detail = 0;
249    state.ctx.host_data = host_data.cast();
250    state.ctx.regs = [0; 32];
251    state.ctx.regs[Reg::SP.index()] = state.memory.stack_pointer();
252    params.write(&mut state.ctx.regs);
253
254    trap::set_guest_region(state.memory.base() as usize, state.memory.size());
255
256    let trampoline: extern "C" fn(*mut VmCtx, *const u8) =
257        unsafe { std::mem::transmute(state.module.trampoline()) };
258    let ctx: *mut VmCtx = &raw mut state.ctx;
259
260    let outcome = trap::protect(|| trampoline(ctx, entry));
261
262    let state = store.state.as_mut().expect("instance is still present");
263    if let Err(fault) = outcome {
264        return Err(Trap::MemoryFault {
265            address: fault.guest,
266        }
267        .into());
268    }
269
270    // A host call records the reason it failed; the compiled code only knows
271    // that it should stop.
272    if let Some(failure) = state.failure.take() {
273        return Err(failure.into());
274    }
275
276    match translator::Trap::from_code(state.ctx.trap) {
277        translator::Trap::None => Ok(R::read(&state.ctx.regs)),
278        translator::Trap::BadIndirectTarget => Err(Trap::BadIndirectTarget {
279            target: state.ctx.detail,
280        }
281        .into()),
282        translator::Trap::Breakpoint => Err(Trap::Breakpoint.into()),
283        translator::Trap::HostCall => Err(Trap::HostCall(anyhow!("host call failed")).into()),
284        translator::Trap::IllegalInstruction => Err(Trap::IllegalInstruction.into()),
285        translator::Trap::Interrupted => Err(Trap::Interrupted.into()),
286    }
287}

289/// Handle an `ecall`.
290///
291/// Compiled code reaches this through [`VmCtx::host_call`]. Returning non-zero
292/// tells the guest to stop; the reason is left in the store.
293extern "C" fn dispatch<T>(ctx: *mut VmCtx) -> u64 {
294    // Read through the raw pointer rather than taking `&mut VmCtx`: the
295    // context lives inside the store, and holding both references at once
296    // would alias.
297    let (host_data, number) = unsafe { ((*ctx).host_data, (*ctx).regs[Reg::A7.index()]) };
298
299    let store = unsafe { &mut *(host_data as *mut Store<T>) };
300    let Some(state) = store.state.as_ref() else {
301        return 1;
302    };
303
304    let hosts = state.hosts.clone();
305    let Some(func) = hosts.get(&number) else {
306        fail(store, Trap::UnknownHostCall(number));
307        return 1;
308    };
309
310    match func(Caller { store }) {
311        Ok(()) => 0,
312        Err(error) => {
313            fail(store, Trap::HostCall(error));
314            1
315        }
316    }
317}
318
319fn fail<T>(store: &mut Store<T>, trap: Trap) {
320    if let Some(state) = store.state.as_mut() {
321        state.failure = Some(trap);
322    }
323}

325/// Stops a running guest.
326///
327/// Cheap to clone and safe to share, so a watchdog can hold one while the guest
328/// runs on another thread.
329#[derive(Clone, Debug)]
330pub struct Interrupt(Arc<AtomicU64>);
331
332impl Interrupt {
333    /// Ask the guest to stop at its next backward edge.
334    ///
335    /// The call returns immediately; the guest stops when it next closes a
336    /// loop, and its pending call fails with [`Trap::Interrupted`].
337    pub fn interrupt(&self) {
338        self.0.store(1, Ordering::Release);
339    }
340
341    /// Withdraw a request, so the guest may run again.
342    pub fn clear(&self) {
343        self.0.store(0, Ordering::Release);
344    }
345
346    /// Whether a stop has been requested.
347    pub fn is_set(&self) -> bool {
348        self.0.load(Ordering::Acquire) != 0
349    }
350}

352/// A host function's view of the guest that called it.
353pub struct Caller<'a, T> {
354    pub(crate) store: &'a mut Store<T>,
355}
356
357impl<T> Caller<'_, T> {
358    /// The embedder's data.
359    pub fn data(&self) -> &T {
360        self.store.data()
361    }
362
363    /// The embedder's data, mutably.
364    pub fn data_mut(&mut self) -> &mut T {
365        self.store.data_mut()
366    }
367
368    /// Read a guest register.
369    pub fn reg(&self, reg: Reg) -> u64 {
370        self.state().ctx.regs[reg.index()]
371    }
372
373    /// Write a guest register.
374    pub fn set_reg(&mut self, reg: Reg, value: u64) {
375        self.state_mut().ctx.regs[reg.index()] = value;
376    }
377
378    /// The guest heap bounds.
379    pub fn heap(&self) -> std::ops::Range<u64> {
380        self.state().memory.heap()
381    }
382
383    /// Read `len` bytes of guest memory at `addr`.
384    pub fn read(&self, addr: u64, len: u64) -> Result<&[u8]> {
385        self.state().memory.read(addr, len)
386    }
387
388    /// Write `data` into guest memory at `addr`.
389    pub fn write(&mut self, addr: u64, data: &[u8]) -> Result<()> {
390        self.state_mut().memory.write(addr, data)
391    }
392
393    /// Borrow the caller again, for passing on to a wrapped closure.
394    pub(crate) fn reborrow(&mut self) -> Caller<'_, T> {
395        Caller { store: self.store }
396    }
397
398    /// The `n`th argument register.
399    pub(crate) fn arg(&self, index: usize) -> u64 {
400        self.reg(Reg::new((Reg::A0.index() + index) as u8))
401    }
402
403    /// Place a host function's results in the argument registers.
404    pub(crate) fn set_results<R: Regs>(&mut self, results: R) {
405        results.write(&mut self.state_mut().ctx.regs);
406    }
407
408    fn state(&self) -> &State<T> {
409        self.store
410            .state
411            .as_ref()
412            .expect("caller implies an instance")
413    }
414
415    fn state_mut(&mut self) -> &mut State<T> {
416        self.store
417            .state
418            .as_mut()
419            .expect("caller implies an instance")
420    }
