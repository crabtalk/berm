15/// The layer, and what the whole artifact is typed as.
16pub const HARNESS: &str = "application/vnd.berm.harness.v1";
17/// The config blob: what the harness says it is.
18pub const MANIFEST: &str = "application/vnd.berm.manifest.v1+json";
19const OCI_MANIFEST: &str = "application/vnd.oci.image.manifest.v1+json";
20/// Accepted only to be refused by name: a registry handed one of these back as
21/// its own error otherwise, which read as a berm bug.
22const OCI_INDEX: &str = "application/vnd.oci.image.index.v1+json";
23
24/// Where a credential comes from. CI has this already; a laptop can export
25/// `gh auth token`. Reading `~/.docker/config.json` is not done here.
26const TOKEN: &str = "GITHUB_TOKEN";

28/// What a token is asked for. A read token is issued anonymously for a public
29/// repository; a write token never is.
30#[derive(Clone, Copy)]
31pub enum Access {
32    Read,
33    Write,
34}
35
36/// A registry, already holding a token for one repository.
37pub struct Registry {
38    base: String,
39    repository: String,
40    http: Client,
41    token: Option<String>,
42}
43
44impl Registry {
45    /// Take the token up front, so a blob upload is never sent twice to answer
46    /// a challenge it could have answered before the body was built.
47    pub fn open(reference: &Reference, access: Access) -> Result<Self> {
48        let scheme = if loopback(&reference.registry) {
49            "http"
50        } else {
51            "https"
52        };
53        let mut registry = Self {
54            base: format!("{scheme}://{}", reference.registry),
55            repository: reference.repository.clone(),
56            http: Client::new(),
57            token: None,
58        };
59
60        let action = match access {
61            Access::Read => "pull",
62            Access::Write => "pull,push",
63        };
64        let scope = format!("repository:{}:{action}", registry.repository);
65        registry.authenticate(&scope)?;
66        Ok(registry)
67    }

69    /// The image, verified against the digest the registry advertised for it.
70    pub fn pull(&self, reference: &str) -> Result<Vec<u8>> {
71        let artifact = self.artifact(reference)?;
72        let layer = layer(&artifact)?;
73        let digest = layer
74            .get("digest")
75            .and_then(Value::as_str)
76            .context("the artifact's layer has no digest")?;
77        self.blob(digest)
78    }
79
80    /// The image's digest and what it says it is, without downloading it.
81    ///
82    /// The config blob is the `.berm.abi` section, so this costs one small GET
83    /// and never touches the image — which is the whole reason it is carried
84    /// there rather than left in the ELF alone. An index is built out of
85    /// exactly these two.
86    pub fn describe(&self, reference: &str) -> Result<(String, Manifest)> {
87        let artifact = self.artifact(reference)?;
88        let image = layer(&artifact)?
89            .get("digest")
90            .and_then(Value::as_str)
91            .context("the artifact's layer has no digest")?
92            .to_owned();
93
94        let config = artifact
95            .get("config")
96            .context("not a harness: the artifact has no config")?;
97
98        let kind = config
99            .get("mediaType")
100            .and_then(Value::as_str)
101            .unwrap_or("");
102        if kind != MANIFEST {
103            bail!("not a harness: its config is {kind:?}");
104        }
105        let digest = config
106            .get("digest")
107            .and_then(Value::as_str)
108            .context("the artifact's config has no digest")?;
109
110        let bytes = self.blob(digest)?;
111        let json = str::from_utf8(&bytes).context("harness manifest is not UTF-8")?;
112        Ok((image, Manifest::parse(json)?))
113    }
114
115    /// Upload the manifest blob, the image, and the artifact manifest naming
116    /// both. Returns the image's digest, which is the layer's.
117    pub fn push(&self, reference: &str, elf: &[u8], manifest: &[u8]) -> Result<String> {
118        let image = self.upload(elf)?;
119        let config = self.upload(manifest)?;
120
121        let artifact = json!({
122            "schemaVersion": 2,
123            "mediaType": OCI_MANIFEST,
124            "artifactType": HARNESS,
125            "config": { "mediaType": MANIFEST, "digest": config, "size": manifest.len() },
126            "layers": [ { "mediaType": HARNESS, "digest": image, "size": elf.len() } ],
127            "annotations": annotations(),
128        });
129
130        let request = self
131            .request(Method::PUT, self.url("manifests", reference))
132            .header(CONTENT_TYPE, OCI_MANIFEST)
133            .body(serde_json::to_vec(&artifact)?);
134        send(request)?;
135        Ok(image)
136    }
137
138    /// The OCI manifest naming the parts, whatever they turn out to be.
139    fn artifact(&self, reference: &str) -> Result<Value> {
140        let request = self
141            .request(Method::GET, self.url("manifests", reference))
142            .header(ACCEPT, format!("{OCI_MANIFEST},{OCI_INDEX}"));
143        send(request)?
144            .json()
145            .context("registry returned a manifest that is not JSON")
146    }
147
148    /// One blob, checked against the digest it is addressed by — before the
149    /// bytes go anywhere. A registry serving the wrong thing is the case this
150    /// exists for, and it is cheap to rule out.
151    fn blob(&self, want: &str) -> Result<Vec<u8>> {
152        let bytes = send(self.request(Method::GET, self.url("blobs", want)))?
153            .bytes()
154            .context("cannot read the blob")?
155            .to_vec();
156
157        let got = digest(&bytes);
158        if got != want {
159            bail!("blob does not match its digest: {want} advertised, {got} received");
160        }
161        Ok(bytes)
162    }
163
164    /// One blob, uploaded monolithically: ask where to put it, then put it.
165    fn upload(&self, bytes: &[u8]) -> Result<String> {
166        let digest = digest(bytes);
167        let uploads = format!("{}/v2/{}/blobs/uploads/", self.base, self.repository);
168        let opened = send(self.request(Method::POST, uploads))?;
169
170        let location = opened
171            .headers()
172            .get(LOCATION)
173            .and_then(|value| value.to_str().ok())
174            .context("registry accepted an upload without saying where to put it")?;
175        // The location may be relative, and may already carry the registry's
176        // own state in a query string.
177        let held = if location.contains('?') { '&' } else { '?' };
178        let url = match location.starts_with("http") {
179            true => format!("{location}{held}digest={digest}"),
180            false => format!("{}{location}{held}digest={digest}", self.base),
181        };
182
183        let request = self
184            .request(Method::PUT, url)
185            .header(CONTENT_TYPE, "application/octet-stream")
186            .body(bytes.to_vec());
187        send(request)?;
188        Ok(digest)
189    }
190
191    /// Ask the registry what it wants, then go and get it.
192    fn authenticate(&mut self, scope: &str) -> Result<()> {
193        let probe = self
194            .http
195            .get(format!("{}/v2/", self.base))
196            .send()
197            .with_context(|| format!("cannot reach {}", self.base))?;
198        if probe.status() != StatusCode::UNAUTHORIZED {
199            return Ok(());
200        }
201
202        let challenge = probe
203            .headers()
204            .get(WWW_AUTHENTICATE)
205            .and_then(|value| value.to_str().ok())
206            .unwrap_or_default()
207            .to_owned();
208        let realm = field(&challenge, "realm")
209            .context("registry asked for a token without saying where from")?;
210
211        // Built rather than encoded: a scope is `repository:org/name:pull`,
212        // and every character in it is one a query value may carry as itself.
213        let service = field(&challenge, "service").unwrap_or_default();
214        let mut request = self
215            .http
216            .get(format!("{realm}?service={service}&scope={scope}"));
217        // GHCR reads the token as the password and ignores the user.
218        if let Ok(token) = env::var(TOKEN) {
219            request = request.basic_auth("berm", Some(token));
220        }
221
222        let answer: Value = send(request)?
223            .json()
224            .context("registry returned a token that is not JSON")?;
225        let token = answer
226            .get("token")
227            .or_else(|| answer.get("access_token"))
228            .and_then(Value::as_str)
229            .context("registry answered without a token")?;
230
231        self.token = Some(token.to_owned());
232        Ok(())
233    }
234
235    fn request(&self, method: Method, url: String) -> RequestBuilder {
236        let request = self.http.request(method, url);
237        match &self.token {
238            Some(token) => request.bearer_auth(token),
239            None => request,
240        }
241    }
242
243    fn url(&self, kind: &str, reference: &str) -> String {
244        format!("{}/v2/{}/{kind}/{reference}", self.base, self.repository)
245    }
246}

248/// Turn a refusal into what the registry said, rather than a status code the
249/// operator then has to go and look up.
250fn send(request: RequestBuilder) -> Result<Response> {
251    let response = request.send().context("cannot reach the registry")?;
252    if response.status().is_success() {
253        return Ok(response);
254    }
255
256    let status = response.status();
257    let body = response.text().unwrap_or_default();
258    match serde_json::from_str::<Value>(&body).ok().and_then(|body| {
259        Some(
260            body.get("errors")?
261                .get(0)?
262                .get("message")?
263                .as_str()?
264                .to_owned(),
265        )
266    }) {
267        Some(message) => bail!("{message}"),
268        None => bail!("registry answered {status}"),
269    }
270}

272/// The one layer a harness has, refusing anything that is not one.
273fn layer(manifest: &Value) -> Result<&Value> {
274    // An index is what a container image looks like, and pointing at one is the
275    // likely mistake here — worth saying so rather than reporting a missing
276    // field.
277    if manifest.get("manifests").is_some() {
278        bail!("not a harness: a multi-platform image, which berm has no use for");
279    }
280
281    let layers = manifest
282        .get("layers")
283        .and_then(Value::as_array)
284        .context("not a harness: the artifact has no layers")?;
285    let [layer] = layers.as_slice() else {
286        bail!(
287            "not a harness: {} layers, where one is the whole of it",
288            layers.len()
289        );
290    };
291
292    let kind = layer.get("mediaType").and_then(Value::as_str).unwrap_or("");
293    if kind != HARNESS {
294        bail!("not a harness: its layer is {kind:?}");
295    }
296    Ok(layer)
297}

299/// Standard OCI keys only. What the harness *is* lives in the config blob, and
300/// a second copy here would be the one that rots.
301fn annotations() -> Value {
302    let mut annotations = Map::new();
303    if let Ok(repository) = env::var("GITHUB_REPOSITORY") {
304        // What makes GHCR show the package against its repository.
305        annotations.insert(
306            "org.opencontainers.image.source".to_owned(),
307            format!("https://github.com/{repository}").into(),
308        );
309    }
310    if let Ok(revision) = env::var("GITHUB_SHA") {
311        annotations.insert(
312            "org.opencontainers.image.revision".to_owned(),
313            revision.into(),
314        );
315    }
316    Value::Object(annotations)
317}

319/// Plain HTTP for a loopback registry, which is the only place one runs
320/// without TLS.
321fn loopback(registry: &str) -> bool {
322    let host = registry.split(':').next().unwrap_or(registry);
323    host == "localhost" || host == "127.0.0.1"
324}

326/// Pull `key="…"` out of a `WWW-Authenticate` challenge.
327fn field(challenge: &str, key: &str) -> Option<String> {
328    let at = challenge.find(&format!("{key}=\""))? + key.len() + 2;
329    let rest = &challenge[at..];
330    Some(rest[..rest.find('"')?].to_owned())
331}

333fn digest(bytes: &[u8]) -> String {
334    format!("sha256:{:x}", Sha256::digest(bytes))
