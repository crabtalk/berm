1//! Reaching an index.
2//!
3//! A clone of the list is a directory and a service that publishes into one is
4//! a URL, so what a caller names takes either shape — the same way `berm
5//! deploy` takes a file or a registry reference.
6
7use crate::Entry;
8use anyhow::{Context, Result, bail};
9use berm_api::Failed;
10use reqwest::blocking::{Client, Response};
11use std::{
12    path::{Path, PathBuf},
13    process::Command,
14};
15
16/// Where to look, when nothing is named.
17const ENV: &str = "BERM_INDEX";
18
19/// The list read when nothing else is named.
20pub const DEFAULT: &str = "https://github.com/crabtalk/berm-index.git";
21
22pub enum Source {
23    /// A clone of the list. Reading it needs no service and no credential.
24    Local(PathBuf),
25    Remote {
26        host: String,
27        http: Client,
28    },
29}
30
31impl Source {
32    /// Blocking, and a `.git` URL clones the first time — so this belongs off
33    /// whatever thread paints, and off the path a keystroke takes.
34    pub fn new(index: Option<&str>) -> Result<Self> {
35        let index = match index {
36            Some(index) => index.to_owned(),
37            None => std::env::var(ENV).unwrap_or_else(|_| DEFAULT.to_owned()),
38        };
39
40        if Path::new(&index).is_dir() {
41            return Ok(Self::Local(PathBuf::from(index)));
42        }
43        // A `.git` URL is a list to copy, not a service to ask. Everything else
44        // that is not already a directory is a service.
45        if index.ends_with(".git") {
46            return Ok(Self::Local(mirror(&index)?));
47        }
48        Ok(Self::Remote {
49            host: index.trim_end_matches('/').to_owned(),
50            http: Client::new(),
51        })
52    }
53
54    pub fn search(&self, term: &str) -> Result<Vec<Entry>> {
55        match self {
56            Self::Local(root) => Ok(crate::Index::load(root)?
57                .search(term)
58                .into_iter()
59                .cloned()
60                .collect()),
61            Self::Remote { host, http } => {
62                let request = http.get(format!("{host}/berm")).query(&[("q", term)]);
63                read(host, request.send())?
64                    .json()
65                    .context("the index returned something that is not a harness list")
66            }
67        }
68    }
69
70    /// The token is whatever the index in front of you wants, and an open one
71    /// wants none — so it rides along when there is one and is not invented
72    /// when there is not.
73    pub fn publish(&self, reference: &str, token: Option<&str>) -> Result<Entry> {
74        let Self::Remote { host, http } = self else {
75            bail!("publishing needs a service to publish to, not a clone of the list");
76        };
77
78        let mut request = http
79            .post(format!("{host}/berm"))
80            .json(&serde_json::json!({ "reference": reference }));
81        if let Some(token) = token {
82            request = request.bearer_auth(token);
83        }
84        read(host, request.send())?
85            .json()
86            .context("the index returned something that is not a harness")
87    }
88}

90/// The local copy of a list, cloned if it is not there yet.
91///
92/// Keyed by the URL so two indexes never share a directory. Cloned shallow —
93/// what matters is the list, not how it came to be — and left alone afterwards,
94/// so a search never waits on a network. `git -C <path> pull` refreshes it.
95fn mirror(url: &str) -> Result<PathBuf> {
96    let home = std::env::var("HOME").context("HOME is not set")?;
97    let path = url
98        .split_once("://")
99        .map_or(url, |(_, rest)| rest)
100        .trim_end_matches(".git");
101    let into = PathBuf::from(home).join(".berm/index").join(path);
102    if into.is_dir() {
103        return Ok(into);
104    }
105
106    let status = Command::new("git")
107        .args(["clone", "--quiet", "--depth", "1", url])
108        .arg(&into)
109        .status()
110        .context("cannot run git, which is what copies an index")?;
111    if !status.success() {
112        bail!("git could not clone {url}");
113    }
114    Ok(into)
115}

117/// Turn a refusal into the message the index sent, rather than a status code
118/// the caller then has to go and look up.
119fn read(host: &str, response: reqwest::Result<Response>) -> Result<Response> {
120    let response = response.with_context(|| format!("cannot reach {host}"))?;
121    if response.status().is_success() {
122        return Ok(response);
123    }
124
125    let status = response.status();
126    match response.json::<Failed>() {
127        Ok(failed) => bail!("{}", failed.error),
128        Err(_) => bail!("{host} answered {status}"),
129    }
130}
