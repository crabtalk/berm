3use berm_api::{Manifest, ToolSpec};
4use serde::{Deserialize, Serialize};
5
6/// What an index records about a harness at one digest.
7///
8/// Keyed by a digest, which is why the tools and usage beside it are safe to
9/// keep: the bytes at a digest can never change, so a copy of what they said
10/// cannot go stale. Re-pushing a tag does not rewrite an entry — it adds one,
11/// and the old line still truthfully describes the old bytes.
12#[derive(Debug, Clone, Serialize, Deserialize)]
13pub struct Entry {
14    /// As published, tag and all.
15    pub reference: String,
16    /// sha256 of the ELF, `sha256:`-prefixed, as the registry addresses it.
17    pub digest: String,
18    /// Whoever the publishing service vouched for, if it vouched for anyone.
19    /// An open index records none, and the reference already names an owner
20    /// that anyone can check against the registry.
21    #[serde(default, skip_serializing_if = "Option::is_none")]
22    pub publisher: Option<String>,
23    pub usage: String,
24    pub tools: Vec<ToolSpec>,
25}
26
27impl Entry {
28    pub fn new(
29        reference: String,
30        digest: String,
31        publisher: Option<String>,
32        manifest: Manifest,
33    ) -> Self {
34        Self {
35            reference,
36            digest,
37            publisher,
38            usage: manifest.usage,
39            tools: manifest.tools,
40        }
41    }
42
43    /// The harness this is a version of — the reference without its tag, which
44    /// is what one file in the index is named for.
45    pub fn key(&self) -> &str {
46        match self.reference.split_once('@') {
47            Some((repository, _)) => repository,
48            None => self
49                .reference
50                .rsplit_once(':')
51                .map_or(self.reference.as_str(), |(repository, _)| repository),
52        }
53    }
54
55    /// Whether `term` appears in anything a person would search by. The tool
56    /// descriptions are here because "which harness reads files" is the
57    /// question being asked, and no tool is named after it.
58    pub fn matches(&self, term: &str) -> bool {
59        let term = term.to_lowercase();
60        self.reference.to_lowercase().contains(&term)
61            || self.usage.to_lowercase().contains(&term)
62            || self.tools.iter().any(|tool| {
63                tool.name.to_lowercase().contains(&term)
64                    || tool.description.to_lowercase().contains(&term)
65            })
66    }
67}
