1//! How a system harness request is laid out.
2//!
3//! Every request is a sequence of length-prefixed byte fields — one rule with
4//! no exceptions, so the host has a single decoder and a system harness taking one
5//! argument is framed the same way as one taking five. The four bytes a
6//! single-field request spends on its own length buy that.
7//!
8//! A reply uses the same layout when it carries more than one blob, which is
9//! why [`fields()`](crate::abi::wire::fields) is here: the framing is the wire,
10//! not the direction.
11//!
12//! Not JSON, because file content is arbitrary bytes and JSON cannot carry
13//! those without base64. A system harness that moves bytes should not pay an
14//! encoding to do it.
15
16use alloc::vec::Vec;
17
18/// Append one length-prefixed field.
19pub fn field(request: &mut Vec<u8>, bytes: &[u8]) {
20    request.extend_from_slice(&(bytes.len() as u32).to_le_bytes());
21    request.extend_from_slice(bytes);
22}
23
24/// Build a request from its fields.
25pub fn request(fields: &[&[u8]]) -> Vec<u8> {
26    let mut request = Vec::new();
27    for bytes in fields {
28        field(&mut request, bytes);
29    }
30    request
31}
32
33/// Split a framed reply back into its fields. `None` if it is malformed,
34/// which for a guest means the host is not the one it was built against.
35pub fn fields(mut framed: &[u8]) -> Option<Vec<&[u8]>> {
36    let mut fields = Vec::new();
37    while !framed.is_empty() {
38        let (header, rest) = framed.split_at_checked(4)?;
39        let length = u32::from_le_bytes(header.try_into().ok()?) as usize;
40        let (field, rest) = rest.split_at_checked(length)?;
41        fields.push(field);
42        framed = rest;
43    }
44    Some(fields)
45}
