1//! The wire between a harness and its host.
2//!
3//! Host calls travel in `a7`, which is a number — but the number is *derived*
4//! from a name rather than assigned (RFC 0205). A system harness is identified by
5//! what it is called, so adding one cannot collide with someone else's
6//! allocation and no registry of integers has to be maintained.
7//!
8//! The same hash is computed host-side in `crates/berm/src/abi.rs`. The two
9//! must agree; if they ever drift, every call traps immediately as an unknown
10//! host call rather than reaching the wrong system harness.
11//!
12//! Whether a call reaches a host at all is `sys`'s business, not this file's.
13//!
14//! Nothing here is for a harness author to read: [`host`] and [`wire`] are
15//! reached by generated code, and the numbers below by this module's own
16//! functions. What an author writes against is the crate root.
17
18use crate::sys;
19
20/// One call path shared by every system harness. Needs a heap: a result whose
21/// size the guest learns at runtime has nowhere else to go.
22#[cfg(feature = "alloc")]
23#[doc(hidden)]
24pub mod host;
25/// Request framing, on the same terms as [`host`].
26#[cfg(feature = "alloc")]
27#[doc(hidden)]
28pub mod wire;
29
30/// Write a UTF-8 message to the host log.
31pub const HOST_LOG: u64 = hash("berm.log");
32/// Byte length of this invocation's argument blob.
33pub const HOST_ARG_LEN: u64 = hash("berm.args.len");
34/// Copy the argument blob into guest memory.
35pub const HOST_ARG_READ: u64 = hash("berm.args.read");
36/// Fail this invocation with a message.
37pub const HOST_FAIL: u64 = hash("berm.fail");
38/// Copy the last system harness call's staged result into guest memory.
39pub const HOST_RESULT_READ: u64 = hash("berm.result.read");
40/// Where this guest's heap starts. Asked for on the first allocation, from
41/// inside the entry the guest is already in.
42pub const HOST_HEAP_START: u64 = hash("berm.heap.start");
43/// How many bytes of it there are.
44pub const HOST_HEAP_SIZE: u64 = hash("berm.heap.size");
45
46/// Set on the length a system harness returns when the staged bytes are an error
47/// message rather than a result. A length never reaches this bit on its own,
48/// so one return value carries both without a second call to ask which.
49pub(crate) const ERROR: u64 = 1 << 63;
50
51/// FNV-1a over the system harness's name, evaluated at compile time.
52pub const fn hash(name: &str) -> u64 {
53    let bytes = name.as_bytes();
54    let mut result: u64 = 0xcbf2_9ce4_8422_2325;
55    let mut at = 0;
56    while at < bytes.len() {
57        result ^= bytes[at] as u64;
58        result = result.wrapping_mul(0x0000_0100_0000_01b3);
59        at += 1;
60    }
61    result
62}

64/// A pointer and length, which is exactly what two result registers hold.
65/// `repr(C)` puts them in `a0` and `a1` under LP64.
66#[repr(C)]
67pub struct Buf {
68    pub ptr: u64,
69    pub len: u64,
70}
71
72impl Buf {
73    /// Point at bytes the host may read after the guest returns. They must
74    /// outlive the call — a static buffer or a leaked allocation, never a
75    /// local.
76    pub fn new(bytes: &[u8]) -> Self {
77        Self {
78            ptr: bytes.as_ptr() as u64,
79            len: bytes.len() as u64,
80        }
81    }
82}
83
84/// Write a line to the host's log.
85pub fn log(message: &str) {
86    sys::call2(HOST_LOG, message.as_ptr() as u64, message.len() as u64);
87}
88
89/// How many bytes this invocation was given.
90pub fn args_len() -> usize {
91    sys::call0(HOST_ARG_LEN) as usize
92}
93
94/// Pull the argument blob into `buffer`, returning the blob's *full* length —
95/// not what fit. A caller that gets back more than it offered was truncated
96/// and must say so rather than acting on half a request.
97pub fn read_args(buffer: &mut [u8]) -> usize {
98    sys::call2(
99        HOST_ARG_READ,
100        buffer.as_mut_ptr() as u64,
101        buffer.len() as u64,
102    ) as usize
103}
104
105/// Report failure. The host marks the invocation an error rather than a
106/// result, which is the difference between a tool that failed and a tool that
107/// returned the word "error".
108pub fn fail(message: &[u8]) -> Buf {
109    sys::call2(HOST_FAIL, message.as_ptr() as u64, message.len() as u64);
110    Buf { ptr: 0, len: 0 }
111}
112
113/// Pull the last system harness call's staged result, returning its *full* length
114/// exactly as [`read_args`] does — the one pattern both use.
115pub(crate) fn read_result(buffer: &mut [u8]) -> usize {
116    sys::call2(
117        HOST_RESULT_READ,
118        buffer.as_mut_ptr() as u64,
119        buffer.len() as u64,
120    ) as usize
121}
