1//! How a system harness call works, once, for all of them.
2//!
3//! A call hands the host a request and gets back a length. The bytes stay
4//! host-side until the guest asks for them, which is the same pull the
5//! argument blob uses (`args.len` then `args.read`) and is here for the same
6//! reason: the host never enters a guest to give it something, and a result
7//! whose size is unknown in advance cannot be measured by running the work
8//! twice.
9//!
10//! Failure travels on the same wire. The high bit of the returned length says
11//! the staged bytes are a message rather than a result, so an error costs no
12//! extra call and cannot be mistaken for an empty success.
13
14use crate::{
15    abi::{ERROR, read_result},
16    sys,
17};
18use alloc::{string::String, vec, vec::Vec};
19
20/// Make one system harness call. `Err` carries whatever the host said went wrong.
21pub fn call(number: u64, request: &[u8]) -> Result<Vec<u8>, String> {
22    let staged = sys::call2(number, request.as_ptr() as u64, request.len() as u64);
23    let failed = staged & ERROR != 0;
24
25    let mut result = vec![0u8; (staged & !ERROR) as usize];
26    let full = read_result(&mut result);
27    if full != result.len() {
28        return Err(String::from("host staged a result of a different length"));
29    }
30
31    if failed {
32        return Err(String::from_utf8_lossy(&result).into_owned());
33    }
34    Ok(result)
35}
