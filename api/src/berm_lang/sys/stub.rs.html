1//! What a host call does when there is no host.
2//!
3//! Off RISC-V this crate is an ordinary library, so a harness author can build
4//! and unit test their handlers natively rather than cross-compiling to see
5//! anything at all. The calls a test can reasonably answer — the argument blob,
6//! the log, the failure channel — are served from [`crate::test`]'s state.
7//! Anything else panics naming the system harness, because a test that reached a
8//! real system harness should say so rather than read a plausible zero.
9
10use crate::abi;
11use std::{cell::RefCell, string::String, vec::Vec};
12
13std::thread_local! {
14    static HOST: RefCell<Host> = const { RefCell::new(Host::new()) };
15}
16
17/// The bits of a host a test can stand in for.
18pub(crate) struct Host {
19    pub args: Vec<u8>,
20    pub logged: Vec<String>,
21    pub failure: Option<String>,
22}
23
24impl Host {
25    const fn new() -> Self {
26        Self {
27            args: Vec::new(),
28            logged: Vec::new(),
29            failure: None,
30        }
31    }
32}
33
34/// Run `f` against the thread's stand-in host.
35pub(crate) fn with<T>(f: impl FnOnce(&mut Host) -> T) -> T {
36    HOST.with(|host| f(&mut host.borrow_mut()))
37}
38
39#[inline]
40pub fn call0(number: u64) -> u64 {
41    match number {
42        abi::HOST_ARG_LEN => with(|host| host.args.len() as u64),
43        _ => no_host(number),
44    }
45}
46
47#[inline]
48pub fn call2(number: u64, a0: u64, a1: u64) -> u64 {
49    match number {
50        abi::HOST_ARG_READ => with(|host| {
51            let taken = host.args.len().min(a1 as usize);
52            // Safety: a0/a1 describe a buffer the caller owns, exactly as the
53            // real host requires. Off-target that buffer is ordinary memory.
54            unsafe { core::ptr::copy_nonoverlapping(host.args.as_ptr(), a0 as *mut u8, taken) };
55            host.args.len() as u64
56        }),
57        abi::HOST_LOG => {
58            with(|host| host.logged.push(read(a0, a1)));
59            0
60        }
61        abi::HOST_FAIL => {
62            with(|host| host.failure = Some(read(a0, a1)));
63            0
64        }
65        _ => no_host(number),
66    }
67}
68
69/// Read a `(ptr, len)` pair the caller passed. Off-target these are real
70/// addresses in this process.
71fn read(ptr: u64, len: u64) -> String {
72    let bytes = unsafe { core::slice::from_raw_parts(ptr as *const u8, len as usize) };
73    String::from_utf8_lossy(bytes).into_owned()
74}
75
76#[cold]
77fn no_host(number: u64) -> ! {
78    panic!("host call {number:#x} has no stand-in; it needs a real host");
79}
