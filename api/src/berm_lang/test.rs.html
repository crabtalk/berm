1//! Run a harness's tools natively, without a toolchain or a daemon.
2//!
3//! A harness is an ELF that a host compiles and enters, which is a slow way to
4//! find out that a handler mishandles an empty string. Off the guest's target
5//! the exports are ordinary functions and the buffers are ordinary memory, so a
6//! test can call one the way the host would — the same argument transfer, the
7//! same buffer limits, the same failure channel — and check what came back.
8//!
9//! ```ignore
10//! #[test]
11//! fn echo_wraps_the_payload() {
12//!     let out = berm_lang::test::call(berm_tool_echo, b"hi").unwrap();
13//!     assert_eq!(out, br#"{"echo":hi}"#);
14//! }
15//! ```
16#![cfg(not(target_arch = "riscv64"))]
17
18use crate::{Buf, sys};
19use std::{
20    string::String,
21    sync::{Mutex, PoisonError},
22    vec::Vec,
23};
24
25/// A harness writes into buffers the macro puts in `.bss`, which on the guest
26/// is safe because one invocation runs at a time. A test binary runs its tests
27/// in parallel, so they take turns here or they write over each other.
28static ONE_AT_A_TIME: Mutex<()> = Mutex::new(());
29
30/// Invoke a tool export with `args`, as the host does.
31///
32/// `Err` is what the harness reported through its failure channel — the same
33/// distinction the host draws between a tool that failed and one that returned
34/// the word "error".
35pub fn call(entry: extern "C" fn() -> Buf, args: &[u8]) -> Result<Vec<u8>, String> {
36    // A failing assertion is a test doing its job, so a panic while holding
37    // this must not poison it into failing every other test.
38    let _turn = ONE_AT_A_TIME.lock().unwrap_or_else(PoisonError::into_inner);
39
40    sys::with(|host| {
41        host.args = args.to_vec();
42        host.logged.clear();
43        host.failure = None;
44    });
45
46    let Buf { ptr, len } = entry();
47
48    if let Some(failure) = sys::with(|host| host.failure.take()) {
49        return Err(failure);
50    }
51    // Safety: the harness returned this pair to be read, and off-target its
52    // buffers are ordinary memory in this process.
53    let bytes = unsafe { core::slice::from_raw_parts(ptr as *const u8, len as usize) };
54    Ok(bytes.to_vec())
55}
56
57/// What the harness logged during the last [`call`].
58pub fn logged() -> Vec<String> {
59    sys::with(|host| host.logged.clone())
60}
