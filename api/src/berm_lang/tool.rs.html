1//! What every tool body does before and after its own work.
2//!
3//! Reading arguments and reporting a failure are the same in every harness,
4//! and a copy per crate is a copy that drifts — one of them starts wording an
5//! error differently, and a model sees two conventions from one runtime.
6#![cfg(feature = "alloc")]
7
8use crate::{Failed, Out};
9use alloc::string::String;
10use core::fmt::Write;
11
12/// Deserialize a tool's arguments, reporting the error to the model rather
13/// than trapping — a malformed call is the model's mistake to fix, not the
14/// harness's to die of.
15///
16/// Arguments land in a struct rather than a dynamic value because the latter
17/// reaches the guest's one unsupported construct, dynamic dispatch, and traps.
18/// See `docs/src/rfcs/0205-berm.md`.
19#[cfg(feature = "args")]
20pub fn parse<T: for<'de> serde_guest::Deserialize<'de>>(
21    args: &[u8],
22    out: &mut Out,
23) -> Result<T, Failed> {
24    match serde_json_guest::from_slice(args) {
25        Ok(parsed) => Ok(parsed),
26        Err(error) => {
27            let _ = write!(out, "invalid arguments: {error}");
28            Err(Failed)
29        }
30    }
31}
32
33/// Turn a system harness's failure into this tool's failure, unchanged. The host
34/// already said what went wrong, and in more detail than a rewording would
35/// keep.
36pub fn system<T>(result: Result<T, String>, out: &mut Out) -> Result<T, Failed> {
37    result.map_err(|error| {
38        out.write(error.as_bytes());
39        Failed
40    })
41}
42
43/// Fail with a message, for a tool that has decided its own arguments are
44/// wrong before reaching a system harness.
45pub fn failed(message: &str, out: &mut Out) -> Result<(), Failed> {
46    out.write(message.as_bytes());
47    Err(Failed)
48}
