1//! Deploying, restoring, and invoking.
2
3use crate::Service;
4use anyhow::{Context, Result, bail};
5use berm::{Berm, Manifest};
6use sha2::{Digest, Sha256};
7use std::{path::PathBuf, sync::Arc};
8
9/// A harness the service is holding: its bytes pinned, its module compiled,
10/// its manifest read.
11pub struct Deployed {
12    pub name: String,
13    /// sha256 of the ELF. Redeploying different bytes under the same name is a
14    /// different harness, and this is what says so.
15    pub digest: String,
16    berm: Berm,
17}
18
19impl Deployed {
20    pub fn manifest(&self) -> &Manifest {
21        self.berm.manifest()
22    }
23}
24
25impl Service {
26    /// Compile `elf`, store it, and make its tools reachable under `name`.
27    ///
28    /// Compile, then disk, then memory — in that order throughout. Compiling
29    /// first keeps a rejected image from leaving behind a file that fails again
30    /// on every restart; writing before publishing keeps the service from
31    /// serving a tool that would vanish when it restarts.
32    pub async fn deploy(&self, name: &str, elf: Vec<u8>) -> Result<Arc<Deployed>> {
33        validate(name)?;
34        let elf = Arc::new(elf);
35        let deployed = self.compile(name.to_owned(), elf.clone()).await?;
36
37        tokio::fs::create_dir_all(self.images())
38            .await
39            .context("failed to open the image directory")?;
40        tokio::fs::write(self.image(name), elf.as_slice())
41            .await
42            .context("failed to store the image")?;
43
44        self.publish(deployed.clone()).await;
45        Ok(deployed)
46    }
47
48    /// Forget a harness and delete its image. `false` if it wasn't deployed.
49    pub async fn undeploy(&self, name: &str) -> Result<bool> {
50        if self.get(name).await.is_none() {
51            return Ok(false);
52        }
53        tokio::fs::remove_file(self.image(name))
54            .await
55            .context("failed to remove the image")?;
56
57        self.deployed.write().await.remove(name);
58        let _ = self.changed.send(());
59        Ok(true)
60    }
61
62    /// Run one tool.
63    ///
64    /// The outer `Result` is the service's — no such harness, no such tool, a
65    /// trap. The inner one is the harness's own reported failure, which is a
66    /// tool result rather than an error.
67    pub async fn call(
68        &self,
69        harness: &str,
70        tool: &str,
71        args: Vec<u8>,
72    ) -> Result<Result<String, String>> {
73        let Some(deployed) = self.get(harness).await else {
74            bail!("no harness named {harness:?} is deployed");
75        };
76        let tool = tool.to_owned();
77        // Entering a guest blocks the thread until the guest returns, which is
78        // not something a runtime worker can afford to do.
79        tokio::task::spawn_blocking(move || deployed.berm.call(&tool, args))
80            .await
81            .context("invocation panicked")?
82    }
83
84    /// Bring back what was deployed before this process started.
85    pub(crate) async fn restore(&self) -> Result<()> {
86        let images = self.images();
87        if !images.is_dir() {
88            return Ok(());
89        }
90
91        let mut entries = tokio::fs::read_dir(&images)
92            .await
93            .context("failed to read the image directory")?;
94        while let Some(entry) = entries.next_entry().await? {
95            let path = entry.path();
96            let Some(name) = path
97                .extension()
98                .filter(|extension| *extension == "elf")
99                .and_then(|_| path.file_stem())
100                .and_then(|stem| stem.to_str())
101            else {
102                continue;
103            };
104
105            let elf = Arc::new(tokio::fs::read(&path).await?);
106            // One unloadable image is not a reason to come up with none of
107            // them: it is reported and skipped, and the rest still serve.
108            match self.compile(name.to_owned(), elf).await {
109                Ok(deployed) => {
110                    tracing::info!(name, digest = %deployed.digest, "restored");
111                    self.publish(deployed).await;
112                }
113                Err(error) => tracing::error!(name, "{error:#}"),
114            }
115        }
116        Ok(())
117    }
118
119    /// Compile an image and read what it claims to be.
120    ///
121    /// Doing this at deploy rather than on first call means a broken image is
122    /// refused by the deploy that introduced it, not on a model's turn — and
123    /// `Berm::load` checks the manifest against the symbol table on the way.
124    async fn compile(&self, name: String, elf: Arc<Vec<u8>>) -> Result<Arc<Deployed>> {
125        let digest = format!("{:x}", Sha256::digest(elf.as_slice()));
126        let engine = self.engine.clone();
127        let berm = tokio::task::spawn_blocking(move || Berm::load(&engine, &elf, &[]))
128            .await
129            .context("compilation panicked")??;
130        Ok(Arc::new(Deployed { name, digest, berm }))
131    }
132
133    /// Make a compiled harness reachable, and tell every session the tool set
134    /// moved under it.
135    async fn publish(&self, deployed: Arc<Deployed>) {
136        self.deployed
137            .write()
138            .await
139            .insert(deployed.name.clone(), deployed);
140        let _ = self.changed.send(());
141    }
142
143    fn images(&self) -> PathBuf {
144        self.root.join("harnesses")
145    }
146
147    fn image(&self, name: &str) -> PathBuf {
148        self.images().join(format!("{name}.elf"))
149    }
150}
151
152/// A name has to survive being half of an MCP tool name and all of a filename.
153/// The dot is the one that matters: dispatch splits `{harness}.{tool}` on the
154/// first one, so a harness carrying a dot would eat its own tool.
155fn validate(name: &str) -> Result<()> {
156    if name.is_empty() {
157        bail!("a harness name cannot be empty");
158    }
159    if !name
160        .chars()
161        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-' || c == '_')
162    {
163        bail!("harness name {name:?} is not lowercase letters, digits, `-` or `_`");
164    }
165    Ok(())
166}
