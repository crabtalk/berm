1//! Guest fault recovery
2//!
3//! Out-of-bounds guest accesses land on guard pages rather than explicit bounds
4//! checks, so recovering from one means catching a signal raised inside
5//! JIT-compiled code and unwinding out of it. See `csrc/trap.c`.
6
7use core::ffi::c_void;
8use std::{cell::Cell, ptr};
9
10unsafe extern "C" {
11    fn rvtime_protect(
12        body: extern "C" fn(*mut c_void) -> bool,
13        payload: *mut c_void,
14        slot: *mut *mut c_void,
15    ) -> bool;
16    fn rvtime_unwind(landing_pad: *mut c_void) -> !;
17    fn rvtime_install(handler: extern "C" fn(i32, *mut libc::siginfo_t, *mut c_void)) -> i32;
18}
19
20thread_local! {
21    /// The innermost `rvtime_protect` frame, or null outside the guest.
22    static LANDING_PAD: Cell<*mut c_void> = const { Cell::new(ptr::null_mut()) };
23    /// Where the fault happened, recorded by the handler before unwinding.
24    static FAULT: Cell<Option<Fault>> = const { Cell::new(None) };
25    /// Signal handlers are per-process, but installation is tracked per thread.
26    static INSTALLED: Cell<bool> = const { Cell::new(false) };
27    /// Base and size of this thread's guest memory, for turning a host fault
28    /// address back into a guest one.
29    static GUEST_REGION: Cell<(usize, u64)> = const { Cell::new((0, 0)) };
30}
31
32/// A guest memory fault.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub struct Fault {
35    /// The signal number, `SIGSEGV` or `SIGBUS`.
36    pub signal: i32,
37
38    /// The faulting address translated back into the guest address space, or
39    /// `None` if it fell outside the reserved window.
40    pub guest: Option<u64>,
41
42    /// The raw host address, for diagnostics.
43    pub host: usize,
44}
45
46/// Record the guest memory region so faults can be reported in guest terms.
47///
48/// Called when a guest is entered on this thread.
49pub fn set_guest_region(base: usize, size: u64) {
50    GUEST_REGION.with(|region| region.set((base, size)));
51}
52
53/// Run `body`, catching any guest memory fault it raises.
54pub fn protect<F, T>(body: F) -> Result<T, Fault>
55where
56    F: FnOnce() -> T,
57{
58    install();
59    FAULT.with(|f| f.set(None));
60
61    // The closure and its result are passed through a single pointer so the C
62    // side stays free of Rust generics.
63    struct Payload<F, T> {
64        body: Option<F>,
65        result: Option<T>,
66    }
67
68    extern "C" fn run<F, T>(payload: *mut c_void) -> bool
69    where
70        F: FnOnce() -> T,
71    {
72        let payload = unsafe { &mut *(payload as *mut Payload<F, T>) };
73        let Some(body) = payload.body.take() else {
74            return false;
75        };
76        payload.result = Some(body());
77        true
78    }
79
80    let mut payload = Payload {
81        body: Some(body),
82        result: None,
83    };
84
85    let completed = LANDING_PAD.with(|slot| unsafe {
86        rvtime_protect(run::<F, T>, &raw mut payload as *mut c_void, slot.as_ptr())
87    });
88
89    match (completed, payload.result) {
90        (true, Some(result)) => Ok(result),
91        _ => Err(FAULT.with(|f| f.get()).unwrap_or(Fault {
92            signal: libc::SIGSEGV,
93            guest: None,
94            host: 0,
95        })),
96    }
97}
98
99/// Install the signal handlers for this thread, once.
100fn install() {
101    INSTALLED.with(|installed| {
102        if installed.get() {
103            return;
104        }
105        let code = unsafe { rvtime_install(handler) };
106        assert_eq!(code, 0, "failed to install guest fault handlers: {code}");
107        installed.set(true);
108    });
109}
110
111/// Record the fault and unwind to the innermost [`protect`].
112///
113/// If there is no landing pad the fault did not come from guest code, so the
114/// handler must not swallow it: restore the default action and let it through.
115extern "C" fn handler(signal: i32, info: *mut libc::siginfo_t, _context: *mut c_void) {
116    let landing_pad = LANDING_PAD.with(|slot| slot.get());
117    if landing_pad.is_null() {
118        unsafe {
119            libc::signal(signal, libc::SIG_DFL);
120        }
121        return;
122    }
123
124    let host = unsafe { (*info).si_addr() } as usize;
125    let (base, size) = GUEST_REGION.with(|region| region.get());
126    let guest = host
127        .checked_sub(base)
128        .map(|offset| offset as u64)
129        .filter(|offset| *offset < size);
130
131    FAULT.with(|f| {
132        f.set(Some(Fault {
133            signal,
134            guest,
135            host,
136        }))
137    });
138
139    unsafe { rvtime_unwind(landing_pad) }
140}
