1//! A disk-backed cache for compiled functions
2//!
3//! Compilation is dominated by code generation — for a 99 KiB guest, decoding
4//! and analysing the ELF is under 1% of the work and Cranelift is the rest. So
5//! the thing worth caching is the generated code, and Cranelift's incremental
6//! cache is keyed on a hash of the CLIF function plus the ISA settings.
7//!
8//! That key is what makes the cache safe to share. It captures the function's
9//! contents rather than its name or address, so two guests containing the same
10//! function reuse one entry, and a change to the target or optimisation level
11//! produces different keys rather than stale code.
12//!
13//! Entries are written atomically, because a daemon may compile the same plugin
14//! from several processes at once.
15
16use anyhow::{Context, Result};
17use cranelift::codegen::incremental_cache::CacheKvStore;
18use std::{
19    borrow::Cow,
20    fs,
21    path::{Path, PathBuf},
22    sync::{
23        Arc,
24        atomic::{AtomicUsize, Ordering},
25    },
26};
27
28/// Compiled functions kept under a directory.
29///
30/// Shared behind an [`Arc`], so every module compiled from one engine
31/// accumulates into the same cache. All of its state is atomic or lives in the
32/// filesystem, which arbitrates concurrent writers.
33pub struct Cache {
34    dir: PathBuf,
35    hits: AtomicUsize,
36    misses: AtomicUsize,
37}
38
39impl Cache {
40    /// Open (and create) a cache directory.
41    pub fn open(dir: impl Into<PathBuf>) -> Result<Self> {
42        let dir = dir.into();
43        fs::create_dir_all(&dir)
44            .with_context(|| format!("failed to create cache directory {}", dir.display()))?;
45        Ok(Cache {
46            dir,
47            hits: AtomicUsize::new(0),
48            misses: AtomicUsize::new(0),
49        })
50    }
51
52    /// The directory entries are kept in.
53    pub fn dir(&self) -> &Path {
54        &self.dir
55    }
56
57    /// Functions served from the cache since it was opened.
58    pub fn hits(&self) -> usize {
59        self.hits.load(Ordering::Relaxed)
60    }
61
62    /// Functions that had to be generated.
63    pub fn misses(&self) -> usize {
64        self.misses.load(Ordering::Relaxed)
65    }
66
67    fn load(&self, key: &[u8]) -> Option<Vec<u8>> {
68        match fs::read(self.path(key)) {
69            Ok(blob) => {
70                self.hits.fetch_add(1, Ordering::Relaxed);
71                Some(blob)
72            }
73            Err(_) => {
74                self.misses.fetch_add(1, Ordering::Relaxed);
75                None
76            }
77        }
78    }
79
80    fn store(&self, key: &[u8], value: Vec<u8>) {
81        // Write to a unique temporary and rename over the target. A partially
82        // written entry would otherwise be indistinguishable from a complete
83        // one, and rename is atomic on every filesystem we care about.
84        let target = self.path(key);
85        let temp = target.with_extension(format!("tmp{}", std::process::id()));
86
87        if fs::write(&temp, value).is_ok() && fs::rename(&temp, &target).is_err() {
88            let _ = fs::remove_file(&temp);
89        }
90    }
91
92    fn path(&self, key: &[u8]) -> PathBuf {
93        use std::fmt::Write;
94        let mut name = String::with_capacity(key.len() * 2);
95        for byte in key {
96            let _ = write!(name, "{byte:02x}");
97        }
98        self.dir.join(name)
99    }
100}

102/// Adapts a shared [`Cache`] to Cranelift's store trait, which takes `&mut self`
103/// to insert. Nothing here needs unique access; the wrapper exists only to
104/// satisfy the signature.
105pub(crate) struct Handle(pub Arc<Cache>);
106
107impl CacheKvStore for Handle {
108    fn get(&self, key: &[u8]) -> Option<Cow<'_, [u8]>> {
109        self.0.load(key).map(Cow::Owned)
110    }
111
112    fn insert(&mut self, key: &[u8], value: Vec<u8>) {
113        self.0.store(key, value);
114    }
115}
