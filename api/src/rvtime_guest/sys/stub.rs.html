1//! What a host call does when there is no host.
2//!
3//! Off RISC-V this crate is an ordinary library linked into someone's tests, so
4//! the wrappers still exist and still type-check — a guest author can build and
5//! test their own logic natively instead of cross-compiling to run anything at
6//! all. A call that actually reaches the boundary panics, naming the number, so
7//! a test that wandered across it says which one rather than reading a
8//! plausible zero.
9
10#[inline]
11pub fn abort() -> ! {
12    panic!("guest aborted outside a guest");
13}
14
15#[cold]
16fn no_host(number: u64) -> ! {
17    panic!("host call {number:#x} outside a guest");
18}
19
20#[inline]
21pub unsafe fn call0(number: u64) -> u64 {
22    no_host(number)
23}
24
25#[inline]
26pub unsafe fn call1(number: u64, _a0: u64) -> u64 {
27    no_host(number)
28}
29
30#[inline]
31pub unsafe fn call2(number: u64, _a0: u64, _a1: u64) -> u64 {
32    no_host(number)
33}
34
35#[inline]
36pub unsafe fn call3(number: u64, _a0: u64, _a1: u64, _a2: u64) -> u64 {
37    no_host(number)
38}
39
40#[inline]
41pub unsafe fn call4(number: u64, _a0: u64, _a1: u64, _a2: u64, _a3: u64) -> u64 {
42    no_host(number)
43}

45#[inline]
46pub unsafe fn call5(number: u64, _a0: u64, _a1: u64, _a2: u64, _a3: u64, _a4: u64) -> u64 {

52pub unsafe fn call6(
53    number: u64,
